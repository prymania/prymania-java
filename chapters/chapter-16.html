บทที่ 16 · Event-driven programming

เหตุการณ์และการตอบสนอง

เชื่อมการกระทำของผู้ใช้ (คลิก พิมพ์ เลือก) เข้ากับโค้ดของเรา ผ่าน listener และ lambda แล้วอัปเดตหน้าจออย่างถูกต้องบน EDT

เมื่อจบบทนี้ นิสิตจะทำได้
  • อธิบาย event source, event object และ listener
  • เขียน ActionListener ด้วย anonymous class และ lambda
  • อ่านค่าจาก component แล้วแสดงผลลัพธ์
  • เก็บ state ของโปรแกรมและอัปเดต UI บน EDT

16. จากหน้าจอนิ่ง สู่หน้าจอที่ตอบสนอง

ในบทที่ 14–15 ปุ่มกดแล้วไม่มีอะไรเกิดขึ้น เพราะยังไม่ได้บอกโปรแกรมว่า “เมื่อคลิกแล้วให้ทำอะไร” บทนี้จะเรียนการเขียนโปรแกรมแบบขับเคลื่อนด้วยเหตุการณ์ (event-driven) ภาพผลลัพธ์ส่วนใหญ่มี 2 ภาพ คือก่อนและหลังการกระทำ ซึ่งได้จากการรันโปรแกรมจริงแล้วจำลองการพิมพ์/คลิก

3 ส่วนของระบบเหตุการณ์
  1. Event source — component ที่เกิดเหตุการณ์ เช่น JButton
  2. Event object — ข้อมูลของเหตุการณ์ เช่น ActionEvent (ใครเป็นต้นเหตุ เมื่อไร)
  3. Listener — โค้ดของเราที่ “ลงทะเบียน” ไว้กับ source และถูกเรียกเมื่อเกิดเหตุการณ์

16.1 Event source และ listener

ลำดับเหตุการณ์เมื่อผู้ใช้คลิกปุ่ม
ผู้ใช้คลิกปุ่ม
    │
    ▼
ระบบปฏิบัติการ → JVM → EDT สร้าง ActionEvent
    │
    ▼
JButton (source) เรียก listener ทุกตัวที่ลงทะเบียนไว้
    │
    ▼
actionPerformed(ActionEvent e) { ...โค้ดของเรา... }   ← ทำงานบน EDT
    │
    ▼
เราเปลี่ยนข้อความ label → Swing วาดหน้าจอใหม่

ตัวอย่าง 16.1.1 · ปุ่มแรกที่ทำงาน: นับจำนวนคลิก พื้นฐาน

แนวคิด: เรียก addActionListener กับปุ่ม และส่งโค้ดที่ต้องทำเมื่อคลิก (ในรูป lambda) — ตัวนับเก็บใน field เพราะต้องจำค่าข้ามการคลิก

ClickCounter.java
import javax.swing.*;

public class ClickCounter {
    // static: lambda แก้ค่าตัวแปรนี้ได้
    private static int count = 0;

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("Counter");
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            JLabel label = new JLabel("Clicks: 0");
            JButton button = new JButton("Click me");
            // ลงทะเบียนงานที่ทำเมื่อคลิกปุ่ม
            button.addActionListener(e -> {
                // โค้ดใน lambda ทำงานทุกครั้งที่คลิก
                count++;
                // อัปเดตข้อความบนหน้าจอ
                label.setText("Clicks: " + count);
            });
            JPanel panel = new JPanel();
            panel.add(button);
            panel.add(label);
            frame.add(panel);
            frame.pack();
            frame.setVisible(true);
        });
    }
}

เริ่มต้น

หลังคลิก 3 ครั้ง

อธิบายทีละขั้น
  1. addActionListener(e -> { ... }) ลงทะเบียนโค้ดไว้กับปุ่ม — ยังไม่ทำงานตอนนี้
  2. main จบ หน้าต่างรอเหตุการณ์
  3. ทุกครั้งที่คลิก โค้ดใน lambda ทำงาน: เพิ่ม count แล้วเปลี่ยนข้อความ label
  4. count ต้องเป็น field (ไม่ใช่ตัวแปร local) เพราะ lambda แก้ตัวแปร local ไม่ได้ และค่าต้องคงอยู่ระหว่างการคลิก

ตัวอย่าง 16.1.2 · หลายปุ่ม หลาย listener ต่อยอด

แนวคิด: แต่ละปุ่มมี listener ของตัวเอง ทุก listener แก้ state ชุดเดียวกัน (value) แล้วเรียกเมธอดอัปเดตหน้าจอ

StepperApp.java
import javax.swing.*;
import java.awt.*;

public class StepperApp {
    private static int value = 0;  // ค่าที่ปุ่มทุกปุ่มใช้ร่วมกัน
    // เก็บเป็น field เพื่อให้ refresh() เข้าถึงได้
    private static JLabel display;

    public static void main(String[] args) {
        // ส่งเมธอด createGui ให้ EDT
        SwingUtilities.invokeLater(StepperApp::createGui);
    }

    private static void createGui() {
        JFrame frame = new JFrame("Stepper");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        display = new JLabel("0", SwingConstants.CENTER);
        display.setFont(new Font("SansSerif", Font.BOLD, 28));
        JButton minus = new JButton("-");
        JButton plus = new JButton("+");
        JButton reset = new JButton("Reset");
        // หลายคำสั่งใน lambda บรรทัดเดียว
        minus.addActionListener(e -> { value--; refresh(); });
        plus.addActionListener(e -> { value++; refresh(); });
        reset.addActionListener(e -> { value = 0; refresh(); });
        JPanel buttons = new JPanel();
        buttons.add(minus);
        buttons.add(plus);
        buttons.add(reset);
        frame.add(display, BorderLayout.CENTER);
        frame.add(buttons, BorderLayout.SOUTH);
        frame.setSize(220, 140);
        frame.setVisible(true);
    }

    private static void refresh() {  // อัปเดตหน้าจอจากค่าปัจจุบัน
        display.setText(String.valueOf(value));
        // ติดลบแสดงสีแดง
        display.setForeground(value < 0 ? Color.RED : Color.BLACK);
    }
}

หลังกด Reset

อธิบายทีละขั้น
  1. state ของโปรแกรมคือ value ตัวเดียว
  2. ทุก listener เปลี่ยน state แล้วเรียก refresh()
  3. refresh() เป็นจุดเดียวที่อัปเดต UI จาก state — ทำให้หน้าจอตรงกับข้อมูลเสมอ
  4. ค่าลบแสดงเป็นสีแดง
ลองคิดก่อน · เมื่อไรโค้ดทำงาน
JButton b = new JButton("Go");
b.addActionListener(e -> System.out.println("clicked"));
System.out.println("listener added");

ข้อความใดถูกพิมพ์ก่อน และ “clicked” ถูกพิมพ์เมื่อไร

ดูคำตอบ

16.2 ActionListener: สามวิธีเขียน

ActionListener เป็น interface ที่มีเมธอดเดียว actionPerformed(ActionEvent e) เขียนได้ 3 แบบ ผลลัพธ์เหมือนกัน: (1) คลาสแยกที่ implements, (2) anonymous class, (3) lambda (สั้นที่สุด นิยมที่สุด)

เปรียบเทียบ 3 แบบ
// (1) คลาสที่ implements ActionListener
class GreetHandler implements ActionListener {
    public void actionPerformed(ActionEvent e) { label.setText("Hi"); }
}
button.addActionListener(new GreetHandler());

// (2) anonymous class — สร้างคลาสไม่มีชื่อ ณ จุดที่ใช้
button.addActionListener(new ActionListener() {
    @Override
    public void actionPerformed(ActionEvent e) { label.setText("Hi"); }
});

// (3) lambda — เขียนเฉพาะ parameter และ body
button.addActionListener(e -> label.setText("Hi"));

ตัวอย่าง 16.2.1 · อ่านค่าจากช่องกรอกแล้วแสดงผล พื้นฐาน

แนวคิด: รูปแบบพื้นฐานที่สุดของ GUI: เมื่อคลิก → getText() จากช่อง → ประมวลผล → setText() ให้ label

GreetApp.java
import javax.swing.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

public class GreetApp {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("Greeter");
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            JTextField nameField = new JTextField(10);
            JButton greet = new JButton("Greet");
            JLabel result = new JLabel("Type your name");
            // anonymous class: วิธีเขียนแบบเต็มก่อนมี lambda
            greet.addActionListener(new ActionListener() {
                @Override
                // เมธอดที่ถูกเรียกเมื่อคลิก
                public void actionPerformed(ActionEvent e) {
                    // อ่านข้อความจากช่องกรอก
                    String name = nameField.getText().trim();
                    result.setText("Hello, " + name + "!");
                }
            });
            JPanel panel = new JPanel();
            panel.add(nameField);
            panel.add(greet);
            panel.add(result);
            frame.add(panel);
            frame.setSize(380, 80);
            frame.setVisible(true);
        });
    }
}

ตัวอย่าง 16.2.2 · แปลงอุณหภูมิ: parse ตัวเลขจากช่องกรอก ต่อยอด

แนวคิด: ค่าจาก JTextField เป็น String เสมอ ต้อง Double.parseDouble ก่อนคำนวณ (บทที่ 9) แล้ว String.format ก่อนแสดง

TempConverter.java
import javax.swing.*;
import java.awt.*;

public class TempConverter {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("C to F");
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            JTextField celsius = new JTextField(6);
            JButton convert = new JButton("Convert");
            JLabel output = new JLabel("? °F");
            convert.addActionListener(e -> {
                // แปลงข้อความเป็นตัวเลขก่อนคำนวณ
                double c = Double.parseDouble(celsius.getText().trim());
                double f = c * 9 / 5 + 32;
                // จัดรูปแบบทศนิยม 1 ตำแหน่ง
                output.setText(String.format("%.1f °F", f));
            });
            JPanel p = new JPanel(new FlowLayout());
            p.add(new JLabel("°C:"));
            p.add(celsius);
            p.add(convert);
            p.add(output);
            frame.add(p);
            frame.pack();
            frame.setVisible(true);
        });
    }
}

  3. แสดงด้วย String.format ทศนิยม 1 ตำแหน่ง

ตัวอย่าง 16.2.3 · listener เดียวหลายปุ่ม: getSource / getActionCommand ประยุกต์

แนวคิด: ปุ่มหลายปุ่มใช้ listener ตัวเดียวได้ แล้วแยกว่าใครกดด้วย e.getActionCommand() (ข้อความบนปุ่ม) หรือ e.getSource() (ออบเจ็กต์ปุ่ม)

QuizButtons.java
import javax.swing.*;
import java.awt.*;
import java.awt.event.ActionListener;

public class QuizButtons {
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            JFrame frame = new JFrame("Quiz");
            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            JLabel question = new JLabel("Which keyword creates an object?  A) class  B) new  C) void");
            JLabel feedback = new JLabel(" ");
            // listener ตัวเดียว ใช้กับหลายปุ่ม
            ActionListener answer = e -> {
                // ข้อความบนปุ่มที่ถูกคลิก

                    feedback.setForeground(Color.RED);
                }
                // getSource() = ปุ่มที่ถูกคลิก
                ((JButton) e.getSource()).setEnabled(false);
            };
            JPanel buttons = new JPanel();
            for (String opt : new String[]{"A", "B", "C"}) {
                JButton b = new JButton(opt);
                // ทุกปุ่มใช้ listener เดียวกัน
                b.addActionListener(answer);
                buttons.add(b);
            }
            JPanel root = new JPanel(new GridLayout(3, 1, 4, 4));
            root.setBorder(BorderFactory.createEmptyBorder(8, 8, 8, 8));
            root.add(question);
            root.add(buttons);
            root.add(feedback);
            frame.setContentPane(root);
            frame.pack();
            frame.setVisible(true);
        });
    }
}

ตัวอย่าง 16.4.1 · แยก model ออกจาก UI: กระปุกออมสิน ประยุกต์

แนวคิด: คลาส PiggyBank (model) มีกฎของตัวเองจากบทที่ 12 ส่วน UI ทำหน้าที่รับคำสั่งและแสดงผล — model ทดสอบได้โดยไม่ต้องมีหน้าจอ

PiggyBankApp.java
import javax.swing.*;
import java.awt.*;

public class PiggyBankApp {
    // ส่วน logic แยกอยู่ในคลาส PiggyBank
    private final PiggyBank bank = new PiggyBank();
    private final JTextField amountField = new JTextField(8);
    private final JLabel balanceLabel = new JLabel();
    private final JLabel message = new JLabel(" ");

    public static void main(String[] args) {
        // สร้างออบเจ็กต์ แล้วเรียก show บน EDT
        SwingUtilities.invokeLater(() -> new PiggyBankApp().show());
    }

    private void show() {
        JFrame frame = new JFrame("Piggy Bank");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        JButton deposit = new JButton("Deposit");
        JButton withdraw = new JButton("Withdraw");
        // true = ฝาก
        deposit.addActionListener(e -> act(true));
        // false = ถอน
        withdraw.addActionListener(e -> act(false));
        JPanel input = new JPanel();
        input.add(new JLabel("Amount:"));
        input.add(amountField);
        input.add(deposit);
        input.add(withdraw);
        JPanel root = new JPanel(new GridLayout(3, 1));
        root.setBorder(BorderFactory.createEmptyBorder(6, 10, 6, 10));
        root.add(balanceLabel);
        root.add(input);
        root.add(message);
        frame.setContentPane(root);
        refresh();
        frame.pack();
        frame.setVisible(true);
    }

    // ทั้งสองปุ่มใช้เมธอดเดียวกัน
    private void act(boolean isDeposit) {
        try {
            // ข้อความที่ไม่ใช่ตัวเลข → NumberFormatException
            double amount = Double.parseDouble(amountField.getText().trim());
            // PiggyBank โยน exception เมื่อข้อมูลผิด
            if (isDeposit) bank.deposit(amount);
            else bank.withdraw(amount);
            message.setText((isDeposit ? "Deposited " : "Withdrew ") + amount);
            message.setForeground(new Color(0, 110, 0));
        // กรอกไม่ใช่ตัวเลข
        } catch (NumberFormatException ex) {
            message.setText("Please enter a number");
            message.setForeground(Color.RED);
        // ข้อความจาก PiggyBank
        } catch (IllegalArgumentException ex) {
            message.setText(ex.getMessage());
            message.setForeground(Color.RED);
        }
        amountField.setText("");
        refresh();  // อัปเดตยอดทุกครั้ง
    }

    private void refresh() {
        balanceLabel.setText(String.format("Balance: %,.2f baht", bank.getBalance()));
    }
}

class PiggyBank {  // คลาส logic ไม่รู้จัก Swing เลย
    private double balance;

    double getBalance() { return balance; }

    void deposit(double amount) {
        // ตรวจข้อมูลแล้วโยน exception
        if (amount <= 0) throw new IllegalArgumentException("Amount must be positive");
        balance += amount;
    }

    void withdraw(double amount) {
        if (amount <= 0) throw new IllegalArgumentException("Amount must be positive");
        // ถอนเกินยอด
        if (amount > balance) throw new IllegalArgumentException("Not enough money (have " + balance + ")");
        balance -= amount;
    }
}

GuessGui.java
import javax.swing.*;
import java.awt.*;

public class GuessGui {
    private int secret = 37;  // สถานะของเกมเก็บใน field
    private int tries = 0;
    private final JTextField input = new JTextField(5);
    private final JButton guess = new JButton("Guess");
    private final JTextArea history = new JTextArea(5, 22);
    private final JLabel status = new JLabel("Guess a number 1-100");

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> new GuessGui().show());
    }

    private void show() {
        JFrame frame = new JFrame("Guess the Number");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        history.setEditable(false);
        JButton restart = new JButton("New game");
        guess.addActionListener(e -> onGuess());  // คลิกปุ่ม
        // กด Enter ในช่องกรอก
        input.addActionListener(e -> onGuess());
        restart.addActionListener(e -> restart());
        JPanel top = new JPanel();
        top.add(input);
        top.add(guess);
        top.add(restart);
        JPanel root = new JPanel(new BorderLayout(4, 4));
        root.setBorder(BorderFactory.createEmptyBorder(8, 8, 8, 8));
        root.add(top, BorderLayout.NORTH);
        root.add(new JScrollPane(history), BorderLayout.CENTER);
        root.add(status, BorderLayout.SOUTH);
        frame.setContentPane(root);
        frame.pack();
        frame.setVisible(true);
    }

    private void onGuess() {
        int g;
        try {  // ตรวจก่อนว่ากรอกตัวเลข
            g = Integer.parseInt(input.getText().trim());
        } catch (NumberFormatException ex) {
            status.setText("Numbers only!");
            return;  // ไม่ใช่ตัวเลข → ไม่นับครั้ง
        }
        tries++;
        if (g == secret) {
            // append ต่อท้ายข้อความเดิม
            history.append(g + " -> correct!\n");
            status.setText("You got it in " + tries + " tries");
            // ทายถูกแล้ว ปิดปุ่มและช่องกรอก
            guess.setEnabled(false);
            input.setEnabled(false);
        } else {
            // ?: เลือกคำใบ้
            String hint = g > secret ? "too high" : "too low";
            history.append(g + " -> " + hint + "\n");
            status.setText("Try #" + (tries + 1));
        }
        input.setText("");
    }

    private void restart() {
        tries = 0;  // เริ่มเกมใหม่: ล้างสถานะทั้งหมด
        history.setText("");
        status.setText("Guess a number 1-100");
        guess.setEnabled(true);
        input.setEnabled(true);
    }
}

import javax.swing.*;
import java.awt.*;
import java.awt.event.ActionListener;

public class WorkingCalculator {
    private double first = 0;  // ตัวเลขแรกที่จำไว้
    private String op = "";    // เครื่องหมายที่รอคำนวณ
    // true = ตัวเลขถัดไปเริ่มใหม่
    private boolean startNew = true;
    private final JTextField display = new JTextField("0", 12);

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> new WorkingCalculator().show());
    }

    private void show() {
        JFrame frame = new JFrame("Calculator");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        display.setEditable(false);
        display.setHorizontalAlignment(JTextField.RIGHT);
        display.setFont(new Font("Monospaced", Font.BOLD, 20));
        ActionListener digit = e -> {  // ปุ่มตัวเลขและจุด
            String d = e.getActionCommand();
            if (startNew) {  // เริ่มตัวเลขใหม่แทนค่าเดิม
                display.setText(d.equals(".") ? "0." : d);
                startNew = false;
            // ห้ามมีจุดทศนิยมซ้ำ
            } else if (!(d.equals(".") && display.getText().contains("."))) {
                display.setText(display.getText() + d);
            }
        };
        ActionListener operator = e -> {  // ปุ่ม + - * /
            // คำนวณค่าที่ค้างอยู่ก่อน
            calculate();
            op = e.getActionCommand();
            startNew = true;
        };
        String[] keys = {"7", "8", "9", "/", "4", "5", "6", "*", "1", "2", "3", "-", "0", ".", "=", "+"};
        JPanel pad = new JPanel(new GridLayout(4, 4, 4, 4));
        for (String k : keys) {
            JButton b = new JButton(k);
            // ปุ่ม = คำนวณแล้วล้างเครื่องหมาย
            if (k.equals("=")) b.addActionListener(e -> { calculate(); op = ""; startNew = true; });
            // ตรวจว่าเป็นเครื่องหมายหรือไม่
            else if ("/*-+".contains(k)) b.addActionListener(operator);
            else b.addActionListener(digit);
            pad.add(b);
        }
        JButton clear = new JButton("C");
        // C ล้างทุกอย่าง
        clear.addActionListener(e -> { first = 0; op = ""; startNew = true; display.setText("0"); });
        JPanel root = new JPanel(new BorderLayout(4, 4));
        root.setBorder(BorderFactory.createEmptyBorder(8, 8, 8, 8));
        root.add(display, BorderLayout.NORTH);
        root.add(pad, BorderLayout.CENTER);
        root.add(clear, BorderLayout.SOUTH);
        frame.setContentPane(root);
        frame.pack();
        frame.setVisible(true);
    }

    private void calculate() {
        // อยู่ในสถานะ error ไม่คำนวณต่อ
        if (display.getText().equals("Error")) return;
        double current = Double.parseDouble(display.getText());
        // ยังไม่มีเครื่องหมาย → จำเป็นตัวเลขแรก
        if (op.isEmpty()) {
            first = current;
            return;
        }
        switch (op) {
            case "+" -> first += current;
            case "-" -> first -= current;
            case "*" -> first *= current;
            case "/" -> {
                // หารด้วย 0 → แสดง Error
                if (current == 0) { display.setText("Error"); op = ""; return; }
                first /= current;
            }
        }
        display.setText(format(first));
    }

    private static String format(double v) {
        // ไม่มีทศนิยมแสดงเป็นจำนวนเต็ม
        return v == Math.rint(v) ? String.valueOf((long) v) : String.valueOf(v);
    }
}

    private final ArrayList<Question> questions = new ArrayList<>();
    // ข้อที่กำลังทำ และคะแนน
    private int index = 0, score = 0;
    private boolean answered = false;
    private final JLabel prompt = new JLabel();
    // ตัวเลือก 4 ข้อ
    private final JRadioButton[] options = new JRadioButton[4];
    // เลือกได้ทีละข้อ
    private final ButtonGroup group = new ButtonGroup();
    private final JLabel feedback = new JLabel(" ");
    private final JButton submit = new JButton("Submit");
    private final JButton next = new JButton("Next");

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> new QuizGui().show());
    }

    private void show() {
        // คำถาม, ตัวเลือก, index คำตอบที่ถูก
        questions.add(new Question("Which keyword creates an object?", new String[]{"class", "new", "void", "static"}, 1));
        questions.add(new Question("Which type stores text?", new String[]{"char", "int", "String", "boolean"}, 2));
        questions.add(new Question("First index of an array?", new String[]{"1", "-1", "length", "0"}, 3));

        JFrame frame = new JFrame("Java Quiz");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        JPanel optionPanel = new JPanel(new GridLayout(4, 1));
        // สร้าง radio button 4 ตัว
        for (int i = 0; i < 4; i++) {
            options[i] = new JRadioButton();
            group.add(options[i]);
            optionPanel.add(options[i]);
        }
        submit.addActionListener(e -> onSubmit());
        // ไปข้อถัดไป
        next.addActionListener(e -> { index++; showQuestion(); });
        JPanel buttons = new JPanel(new FlowLayout(FlowLayout.RIGHT));
        buttons.add(submit);
        buttons.add(next);
        JPanel south = new JPanel(new BorderLayout());
        south.add(feedback, BorderLayout.WEST);
        south.add(buttons, BorderLayout.EAST);
        JPanel root = new JPanel(new BorderLayout(6, 6));
        root.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));
        root.add(prompt, BorderLayout.NORTH);
        root.add(optionPanel, BorderLayout.CENTER);
        root.add(south, BorderLayout.SOUTH);
        frame.setContentPane(root);
        showQuestion();  // แสดงข้อแรก
        frame.setSize(420, 230);
        frame.setVisible(true);
    }

    private void showQuestion() {
        Question q = questions.get(index);  // คำถามข้อปัจจุบัน
        prompt.setText("Q" + (index + 1) + "/" + questions.size() + ": " + q.text);
        // ใส่ข้อความตัวเลือกของข้อนี้
        for (int i = 0; i < 4; i++) options[i].setText(q.choices[i]);
        // ล้างตัวเลือกที่เลือกไว้
        group.clearSelection();
        answered = false;
        feedback.setText(" ");
        submit.setEnabled(true);
        next.setEnabled(false);
    }

    private void onSubmit() {
        int chosen = -1;
        // หาตัวเลือกที่ถูกเลือก
        for (int i = 0; i < 4; i++) if (options[i].isSelected()) chosen = i;
        // ยังไม่ได้เลือก
        if (chosen == -1) {
            feedback.setText("Choose an answer first");
            return;
        }
        Question q = questions.get(index);
        if (chosen == q.answer) {  // เทียบกับ index คำตอบ
            score++;
            feedback.setText("Correct!");
            feedback.setForeground(new Color(0, 128, 0));
        } else {
            feedback.setText("Wrong - answer: " + q.choices[q.answer]);
            feedback.setForeground(Color.RED);
        }
        answered = true;
        // ตอบแล้วส่งซ้ำไม่ได้
        submit.setEnabled(false);
        // ยังไม่ใช่ข้อสุดท้าย
        if (index < questions.size() - 1) {
            next.setEnabled(true);
        } else {
            // ข้อสุดท้าย → แสดงคะแนนรวม
            feedback.setText(feedback.getText() + "   Final score: " + score + "/" + questions.size());
        }
    }
}

class Question {  // เก็บข้อมูลคำถามหนึ่งข้อ
    final String text;
    final String[] choices;
    final int answer;

    Question(String text, String[] choices, int answer) {
        this.text = text;
        this.choices = choices;
        this.answer = answer;
    }
}
