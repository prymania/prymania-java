บทที่ 8 · เก็บข้อมูลหลายค่าเป็นชุด

อาเรย์และชุดข้อมูล

เก็บข้อมูลชนิดเดียวกันหลายค่าในโครงสร้างเดียว แล้วใช้ index และลูปเข้าถึงสมาชิกทีละตัว

เมื่อจบบทนี้ นิสิตจะทำได้
  • ประกาศและกำหนดค่าอาเรย์
  • ใช้ index และ length อย่างถูกต้อง
  • วนอ่าน รวม และค้นหาค่าสมาชิก
  • อ่านข้อมูลแบบแถวและคอลัมน์ และส่งอาเรย์ให้เมธอด

8. มองข้อมูลหลายค่าเป็นชุดเดียว

ถ้าต้องเก็บคะแนนนักศึกษา 100 คน การสร้างตัวแปร 100 ตัว (score1, score2, …) ทำให้จัดการยากและวนลูปไม่ได้ อาเรย์ (array) ให้เราใช้ชื่อตัวแปรเดียวและเลข index เพื่อเลือกสมาชิกแต่ละตำแหน่ง

8.1 สร้างและเข้าถึงอาเรย์

อาเรย์คือกล่องที่แบ่งเป็นช่องเรียงกัน ทุกช่องเก็บข้อมูลชนิดเดียวกัน แต่ละช่องมีเลขประจำตำแหน่งเรียกว่า index ซึ่งเริ่มที่ 0 และอาเรย์มีขนาดคงที่หลังสร้าง

ตัวอย่าง 8.1.1อ่านและแก้ค่าในแต่ละช่อง

ArrayBasics.java
public class ArrayBasics {
    public static void main(String[] args) {
        int[] scores = {80, 75, 92, 88};       // สร้างพร้อมใส่ค่า 4 ช่อง

        System.out.println(scores[0]);           // 80    ช่องแรกคือ index 0
        System.out.println(scores[3]);           // 88    ช่องสุดท้ายคือ index 3
        System.out.println(scores.length);       // 4     length ไม่มีวงเล็บ

        scores[1] = 100;                         // เขียนทับช่อง index 1
        scores[2] += 5;                          // 92 + 5
        System.out.println(scores[1]);           // 100
        System.out.println(scores[2]);           // 97
        System.out.println(scores[scores.length - 1]);   // 88  ช่องสุดท้ายเสมอ
    }
}

รับค่าจากผู้ใช้ใส่อาเรย์

ให้ผู้ใช้บอกขนาดก่อน แล้วใช้ลูปอ่านค่าลงทีละช่อง ตั้งแต่ index 0 ถึง length - 1 รูปแบบนี้ใช้บ่อยที่สุด

ตัวอย่าง 8.1.3อ่านอุณหภูมิ n วัน

ReadIntoArray.java
import java.util.Scanner;

public class ReadIntoArray {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("How many days? ");
        int n = input.nextInt();
        double[] temps = new double[n];                 // ขนาดมาจากผู้ใช้

        System.out.print("Temperatures: ");
        for (int i = 0; i < temps.length; i++) {
            temps[i] = input.nextDouble();              // ใส่ทีละช่อง
        }
        for (int i = 0; i < temps.length; i++) {
            System.out.println("Day " + (i + 1) + ": " + temps[i]);   // คนอ่านนับจาก 1
        }
    }
}

index เกินขอบเขต

index ที่ใช้ได้คือ 0 ถึง length - 1 เท่านั้น ถ้าเขียนเงื่อนไขลูปเป็น i <= scores.length รอบสุดท้ายจะอ่านช่องที่ไม่มี โปรแกรมหยุดด้วย ArrayIndexOutOfBoundsException ใช้ i < scores.length เสมอ

ดูตัวอย่าง: error เมื่อ index เกิน

ทดลอง error 8.1.4index เกินขอบเขต

OutOfBounds.java
public class OutOfBounds {
    public static void main(String[] args) {
        int[] scores = {80, 75, 92};
        for (int i = 0; i <= scores.length; i++) {
            System.out.println(scores[i]);
        }
    }
}

8.2 วนอ่าน รวม และค้นหาสมาชิก

อ่านอาเรย์ทุกช่องด้วยลูปได้ 2 แบบ เลือกตามว่าต้องใช้ index หรือไม่

for ปกติfor-each
รูปแบบfor (int i = 0; i < a.length; i++)for (int x : a)
รู้ indexได้ไม่ได้
แก้ค่าในช่องได้ (a[i] = ...)ไม่ได้
ใช้เมื่อต้องรู้ตำแหน่ง หรือต้องแก้ค่าอ่านอย่างเดียว

ตัวอย่าง 8.2.1วนอ่าน รวม และนับ

ArrayLoop.java
public class ArrayLoop {
    public static void main(String[] args) {
        int[] scores = {72, 45, 88, 91, 50, 39, 67};

        // for ปกติ: ได้ทั้ง index และค่า
        for (int i = 0; i < scores.length; i++) {
            System.out.print(i + ":" + scores[i] + " ");
        }
        System.out.println();

        // for-each: อ่านทีละค่า เหมาะกับการรวมและนับ
        int sum = 0, passed = 0;
        for (int s : scores) {
            sum += s;                  // สะสมผลรวม
            if (s >= 50) passed++;     // นับเฉพาะที่ผ่าน
        }
        double avg = (double) sum / scores.length;     // หารด้วย length ไม่ใช่เลข 7
        System.out.printf("sum %d, avg %.2f, passed %d/%d%n", sum, avg, passed, scores.length);
    }
}

หาค่ามากสุด น้อยสุด และตำแหน่ง

สมมติให้ช่องแรกเป็นทั้งค่ามากสุดและน้อยสุด แล้วเทียบกับช่องที่เหลือ เก็บ index ไว้แทนค่า จะได้รู้ทั้งตำแหน่งและค่า

ตัวอย่าง 8.2.2วันที่ขายดีที่สุดและแย่ที่สุด

FindMax.java
public class FindMax {
    public static void main(String[] args) {
        double[] sales = {1200.5, 980.0, 1530.25, 760.0, 1490.0};
        int maxIndex = 0, minIndex = 0;            // เริ่มจากช่องแรก

        for (int i = 1; i < sales.length; i++) {   // เริ่ม i = 1
            if (sales[i] > sales[maxIndex]) maxIndex = i;
            if (sales[i] < sales[minIndex]) minIndex = i;
        }
        System.out.println("Best day: " + (maxIndex + 1) + " (" + sales[maxIndex] + ")");   // วันที่ 3
        System.out.println("Worst day: " + (minIndex + 1) + " (" + sales[minIndex] + ")");  // วันที่ 4
    }
}

ค้นหาแบบเชิงเส้น (linear search)

ไล่ตรวจทีละช่องจนเจอ ใช้ค่า -1 แทนความหมายว่า “ยังไม่เจอ” และ break ออกจากลูปทันทีที่เจอ

ตัวอย่าง 8.2.3ค้นหาชื่อในรายชื่อ

LinearSearch.java
import java.util.Scanner;

public class LinearSearch {
    public static void main(String[] args) {
        String[] students = {"JHope", "Suga", "RM", "Jungkook", "Mark"};
        Scanner input = new Scanner(System.in);
        System.out.print("Search name: ");
        String target = input.next();

        int found = -1;                                     // -1 = ยังไม่เจอ
        for (int i = 0; i < students.length; i++) {
            if (students[i].equalsIgnoreCase(target)) {
                found = i;
                break;                                      // เจอแล้ว หยุดค้น
            }
        }
        if (found >= 0) System.out.println(target + " is at index " + found);
        else System.out.println(target + " not found");
    }
}

= ไม่ได้คัดลอกอาเรย์

ตัวแปรอาเรย์เก็บที่อยู่ (reference) ของข้อมูล ไม่ได้เก็บข้อมูลเอง b = a จึงทำให้ทั้งสองชื่อชี้ข้อมูลชุดเดียวกัน

ตัวอย่าง 8.2.4ชี้ชุดเดียวกัน vs สำเนาจริง

ArrayReference.java
import java.util.Arrays;

public class ArrayReference {
    public static void main(String[] args) {
        int[] a = {1, 2, 3};
        int[] b = a;              // b ชี้ข้อมูลชุดเดียวกับ a
        int[] c = a.clone();      // c เป็นสำเนาใหม่
        b[0] = 99;                // แก้ผ่าน b

        System.out.println(Arrays.toString(a));        // [99, 2, 3]  a เปลี่ยนด้วย!
        System.out.println(Arrays.toString(c));        // [1, 2, 3]   สำเนาไม่เปลี่ยน
        System.out.println(a == b);                    // true   ชุดเดียวกัน
        System.out.println(Arrays.equals(a, c));       // false  ค่าในช่องต่างกันแล้ว
    }
}

8.3 อาเรย์สองมิติ

อาเรย์สองมิติเก็บข้อมูลแบบตาราง (แถว × คอลัมน์) อ้างช่องด้วย ชื่อ[แถว][คอลัมน์] โดย ชื่อ.length คือจำนวนแถว และ ชื่อ[r].length คือจำนวนคอลัมน์ของแถว r

Grid2D.java
public class Grid2D {
    public static void main(String[] args) {
        int[][] scores = {
            {80, 75, 90},      // แถว 0
            {60, 85, 70}       // แถว 1
        };
        System.out.println(scores.length);      // 2    จำนวนแถว
        System.out.println(scores[0].length);   // 3    จำนวนคอลัมน์
        System.out.println(scores[1][2]);       // 70   แถว 1 คอลัมน์ 2

        for (int r = 0; r < scores.length; r++) {             // ลูปนอก: แถว
            for (int c = 0; c < scores[r].length; c++) {      // ลูปใน: คอลัมน์
                System.out.printf("%4d", scores[r][c]);
            }
            System.out.println();                             // จบแถว ขึ้นบรรทัดใหม่
        }
    }
}

            {78, 85, 90},

            {95, 88, 92}
        };

        for (String sub : subjects) System.out.printf("%6s", sub);
        System.out.printf("%7s%n", "Total");

            int rowSum = 0;                              // เริ่มใหม่ทุกแถว
            System.out.printf("%-6s", students[r]);
            for (int c = 0; c < s[r].length; c++) {
                System.out.printf("%6d", s[r][c]);
                rowSum += s[r][c];
            }
            System.out.printf("%7d%n", rowSum);
        }

        System.out.printf("%-6s", "Avg");
        for (int c = 0; c < subjects.length; c++) {      // ลูปนอกเป็นคอลัมน์
            int colSum = 0;
            for (int r = 0; r < s.length; r++) colSum += s[r][c];
            System.out.printf("%6.1f", (double) colSum / s.length);
        }
        System.out.println();
    }

8.4 อาเรย์กับเมธอด และคลาส Arrays

ส่งอาเรย์ให้เมธอดได้เหมือนตัวแปรทั่วไป แต่สิ่งที่ส่งไปคือที่อยู่ของอาเรย์ ถ้าเมธอดแก้ค่าในช่อง ผู้เรียกจะเห็นการเปลี่ยนแปลงด้วย (ต่างจาก int ในบทที่ 7)

ตัวอย่าง 8.4.1เมธอดที่ใช้ได้กับอาเรย์ทุกขนาด

ArrayMethods.java
public class ArrayMethods {
    public static void main(String[] args) {
        int[] a = {4, 9, 1, 7};
        int[] b = {10, 20};
        System.out.println(sum(a) + " " + max(a));     // 21 9
        System.out.println(sum(b) + " " + max(b));     // 30 20
        doubleAll(b);
        System.out.println(b[0] + " " + b[1]);         // 20 40   b เปลี่ยนจริง
    }

    static int sum(int[] values) {          // รับอาเรย์เป็น parameter
        int total = 0;
        for (int v : values) total += v;
        return total;
    }

    static int max(int[] values) {
        int best = values[0];
        for (int v : values) if (v > best) best = v;
        return best;
    }

    static void doubleAll(int[] values) {   // แก้ช่องในอาเรย์ที่ส่งมา
        for (int i = 0; i < values.length; i++) values[i] *= 2;
    }
}

import java.util.Arrays;

public class ArraysUtility {
    public static void main(String[] args) {
        int[] nums = {42, 7, 19, 3, 25};
        int[] backup = Arrays.copyOf(nums, nums.length);   // สำรองก่อนเรียง
        Arrays.sort(nums);                                  // เรียงน้อยไปมาก

        System.out.println(Arrays.toString(nums));          // [3, 7, 19, 25, 42]
        System.out.println(Arrays.toString(backup));        // [42, 7, 19, 3, 25]
        System.out.println(nums[nums.length / 2]);          // 19   ค่ากลาง

        String[] names = {"Jungkook", "Suga", "JHope"};
        Arrays.sort(names);                                 // เรียงตามตัวอักษร
        System.out.println(Arrays.toString(names));         // [JHope, Jungkook, Suga]

        int[] filled = new int[5];
        Arrays.fill(filled, 7);
        System.out.println(Arrays.toString(filled));        // [7, 7, 7, 7, 7]
        int[] bigger = Arrays.copyOf(backup, 7);            // ช่องที่เกินเป็น 0
        System.out.println(Arrays.toString(bigger));        // [42, 7, 19, 3, 25, 0, 0]
    }
}

BubbleSort.java
import java.util.Arrays;

public class BubbleSort {
    public static void main(String[] args) {
        int[] a = {5, 2, 9, 1, 6};
        System.out.println("start  " + Arrays.toString(a));
        for (int pass = 0; pass < a.length - 1; pass++) {           // n - 1 รอบ
            for (int i = 0; i < a.length - 1 - pass; i++) {         // ช่องท้ายเรียงแล้ว ไม่ต้องเทียบ
                if (a[i] > a[i + 1]) {                              // คู่นี้เรียงผิด → สลับ
                    int temp = a[i];
                    a[i] = a[i + 1];
                    a[i + 1] = temp;
                }
            }
            System.out.println("pass " + (pass + 1) + " " + Arrays.toString(a));
        }
    }
}

เขียนเมธอด static int[] merge(int[] a, int[] b) ที่รับอาเรย์สองชุดที่เรียงจากน้อยไปมากแล้ว และคืนอาเรย์ใหม่ที่มีสมาชิกทั้งหมดเรียงจากน้อยไปมาก ห้ามใช้ Arrays.sort ให้ใช้ตัวชี้ (index) สองตัวเดินไปพร้อมกัน

สิ่งที่ต้องทำ
  1. อาเรย์ผลลัพธ์ขนาด a.length + b.length
  2. เทียบ a[i] กับ b[j] ตัวที่น้อยกว่าใส่ผลลัพธ์ แล้วขยับตัวชี้นั้น
  3. เมื่ออาเรย์หนึ่งหมด ให้คัดลอกส่วนที่เหลือของอีกอาเรย์
  4. ทดสอบกรณีอาเรย์หนึ่งว่าง
ตัวอย่างผลลัพธ์
[1, 2, 3, 4, 7, 8, 10, 12, 15]
[2, 3, 8, 12, 15]

Exercise 8.10เกม Tic-Tac-Toe: ตรวจผู้ชนะ★★★ ท้าทาย

กระดาน 3×3 เก็บใน char[][] board (ค่า 'X', 'O' หรือ '.') ให้ผู้ใช้ป้อนกระดาน 3 บรรทัด (บรรทัดละ 3 ตัวอักษร) แล้วเขียนเมธอด static char winner(char[][] b) คืน 'X' หรือ 'O' ถ้ามีผู้ชนะ (เรียงกัน 3 ตัวในแถว คอลัมน์ หรือทแยง) หรือ '-' ถ้ายังไม่มี จากนั้นแสดงกระดานและผล

สิ่งที่ต้องทำ
  1. อ่านแต่ละบรรทัดด้วย next() แล้วใช้ charAt(c) เติมลงอาเรย์
  2. ตรวจ 3 แถว 3 คอลัมน์ด้วยลูป และทแยง 2 เส้น
  3. ช่องว่าง '.' ไม่นับเป็นผู้ชนะ
  4. ถ้าไม่มีผู้ชนะและไม่มี '.' เหลือ ให้แสดง Draw
