บทที่ 6 · ให้โปรแกรมทำงานซ้ำ

การทำซ้ำด้วยลูป

ลดคำสั่งที่ต้องเขียนซ้ำ โดยกำหนดว่าจะทำซ้ำเมื่อใด ต้องปรับค่าอะไร และจะออกจากลูปอย่างไร

เมื่อจบบทนี้ นิสิตจะทำได้
  • เขียน while และ for ได้ครบองค์ประกอบ
  • เลือก do-while เมื่อต้องทำอย่างน้อยหนึ่งครั้ง
  • ใช้ลูปซ้อนกับรูปแบบข้อมูล
  • ควบคุมและป้องกันลูปไม่รู้จบ

6. ทำซ้ำโดยไม่เขียนคำสั่งเดิมหลายชุด

สมมติต้องพิมพ์คะแนนของนักศึกษา 100 คน ถ้าไม่มีลูปเราต้องเขียนคำสั่งซ้ำ 100 ชุด ซึ่งยาว แก้ยาก และพลาดได้ง่าย ลูปช่วยให้เขียนขั้นตอนครั้งเดียวแล้วให้โปรแกรมทำซ้ำตามเงื่อนไข

องค์ประกอบ 4 อย่างของลูปที่ดี
  1. ค่าเริ่มต้น (initialization) — ตัวนับหรือตัวแปรควบคุมเริ่มที่ค่าใด
  2. เงื่อนไข (condition) — ทำต่อขณะที่อะไรเป็นจริง
  3. งานที่ทำซ้ำ (body) — สิ่งที่ทำในแต่ละรอบ
  4. การอัปเดต (update) — เปลี่ยนค่าที่ทำให้เงื่อนไขเข้าใกล้ false ถ้าขาดข้อนี้จะเกิดลูปไม่รู้จบ

6.1 while: ทำซ้ำขณะเงื่อนไขเป็นจริง

while ตรวจเงื่อนไขก่อนเข้าทุกรอบ ถ้าจริงทำ body แล้วกลับไปตรวจใหม่ ถ้าเท็จออกจากลูป เหมาะเมื่อยังไม่รู้จำนวนรอบแน่นอน

Java
while (condition) {
    // คำสั่งที่ทำซ้ำ
    // เปลี่ยนค่าที่มีผลต่อ condition
}

ตัวอย่าง 6.1.1นับ 1 ถึง 5

CountUp.java
public class CountUp {
    public static void main(String[] args) {
        int number = 1;                    // ค่าเริ่มต้น
        while (number <= 5) {              // ตรวจก่อนเข้าทุกรอบ
            System.out.println(number);
            number++;                      // อัปเดต: ขาดบรรทัดนี้จะวนไม่รู้จบ
        }
        System.out.println("After loop: number = " + number);   // 6 ไม่ใช่ 5
    }
}

ค่าเริ่มต้น → เงื่อนไข → งานที่ทำซ้ำ → การอัปเดตที่ทำให้เงื่อนไขเข้าใกล้ false ถ้าเงื่อนไขเป็น false ตั้งแต่แรก body จะไม่ทำเลยสักรอบ

สะสมผลรวม (accumulator)

รูปแบบที่ใช้บ่อยที่สุด: ตัวแปรผลรวมเริ่มที่ 0 ก่อนลูป แล้วบวกค่าของแต่ละรอบเข้าไป

ตัวอย่าง 6.1.21 + 2 + ... + 5

SumToN.java
public class SumToN {
    public static void main(String[] args) {
        int n = 5;
        int number = 1;
        int sum = 0;                                       // เริ่มที่ 0
        while (number <= n) {
            sum += number;                                 // 1, 3, 6, 10, 15
            System.out.println("add " + number + " -> sum = " + sum);
            number++;
        }
        System.out.println("1 + ... + " + n + " = " + sum);
    }
}

ตัวอย่าง 6.1.3นับหลักและรวมหลักของตัวเลข

CountDigits.java
import java.util.Scanner;

public class CountDigits {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Number: ");
        int n = input.nextInt();
        int digits = 0, digitSum = 0;
        int temp = n;                       // ใช้สำเนา เก็บ n ไว้แสดงผล
        while (temp > 0) {
            digitSum += temp % 10;          // หลักขวาสุด: 5, 1, 2, 7, 0, 9
            temp /= 10;                     // ตัดหลักขวาสุดทิ้ง
            digits++;
        }
        System.out.println(n + " has " + digits + " digits, digit sum = " + digitSum);
    }
}

6.2 for: เมื่อรู้จำนวนรอบหรือตัวนับชัดเจน

for รวมค่าเริ่ม เงื่อนไข และการอัปเดตไว้ในบรรทัดเดียว อ่านแล้วรู้จำนวนรอบทันที

Java
for (int i = 1; i <= 3; i++) {    // ค่าเริ่ม ; เงื่อนไข ; อัปเดต
    // loop body
}

ตัวอย่าง 6.2.1ทำนายรอบก่อนรัน

ForFlow.java
public class ForFlow {
    public static void main(String[] args) {
        int sum = 0;
        for (int i = 1; i <= 3; i++) {             // i = 1, 2, 3 แล้วออกเมื่อ i = 4
            sum += i;
            System.out.println("i=" + i + ", sum=" + sum);
        }
        System.out.println("After loop, sum=" + sum);   // ใช้ i ตรงนี้ไม่ได้ (ประกาศใน for)
    }
}

ใช้ for กับข้อมูล n ค่า

รู้จำนวนข้อมูลล่วงหน้า จึงวนด้วย for ได้ และใช้รูปแบบ “ติดตามค่าที่ดีที่สุด”: ตั้งค่าเริ่มจากข้อมูลตัวแรก แล้วเทียบกับตัวถัดไปทีละตัว

ตัวอย่าง 6.2.3ค่ามาก น้อย และค่าเฉลี่ย

MinMaxAverage.java
import java.util.Scanner;

public class MinMaxAverage {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("How many scores? ");
        int n = input.nextInt();
        System.out.print("Scores: ");
        int first = input.nextInt();
        int max = first, min = first, sum = first;       // ค่าแรกเป็นจุดตั้งต้น
        for (int i = 2; i <= n; i++) {                   // เริ่ม 2 เพราะอ่านค่าแรกแล้ว
            int score = input.nextInt();
            sum += score;
            if (score > max) max = score;
            if (score < min) min = score;
        }
        System.out.printf("Max=%d Min=%d Avg=%.2f%n", max, min, (double) sum / n);
    }
}

import java.util.Scanner;

public class SentinelSum {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int sum = 0, count = 0;
        System.out.print("Number (0 to stop): ");
        int number = input.nextInt();                    // อ่านค่าแรกก่อนเข้าลูป
        while (number != 0) {
            sum += number;                               // 12, 19, 49
            count++;
            System.out.print("Number (0 to stop): ");
            number = input.nextInt();                    // อ่านค่าถัดไปท้ายลูป

        System.out.println("Count = " + count + ", Sum = " + sum);
        if (count > 0) {                                 // กันหารด้วย 0
            System.out.printf("Average = %.2f%n", (double) sum / count);
        }
    }
}

            if (number % 2 == 0) {

        }

    }
}

ใน for ส่วน update ยังทำงานหลัง continue แต่ใน while ถ้า i++ อยู่หลัง continue จะถูกข้าม เช่น if (i % 2 == 0) continue; ก่อน i++ จะค้างที่ i = 2 ตลอดไป แก้โดยย้าย i++ ขึ้นไปก่อน หรือใช้ for

PrimesUpTo50.java
public class PrimesUpTo50 {
    public static void main(String[] args) {
        int count = 0;
        for (int n = 2; n <= 50; n++) {
            boolean isPrime = true;                  // สมมติว่าเป็นก่อน
            for (int d = 2; d * d <= n; d++) {       // d ≤ √n พอ
                if (n % d == 0) {
                    isPrime = false;
                    break;                           // ออกเฉพาะลูป d
                }
            }
            if (isPrime) {
                System.out.print(n + " ");
                count++;
            }
        }
        System.out.println();
        System.out.println("Total primes: " + count);
    }
}

Exercise 6.9ตารางสูตรคูณ 2 มิติ★★★ ท้าทาย

สิ่งที่ต้องทำ
  1. แถวแรกเป็นหัวคอลัมน์ 1..n โดยมุมซ้ายบนเป็นช่องว่าง

Exercise 6.12จำลองเงินฝากดอกเบี้ยทบต้น★★★ ท้าทาย

รับเงินต้น อัตราดอกเบี้ยต่อปี (%) และเงินที่ต้องการมี (เป้าหมาย) ดอกเบี้ยทบต้นปีละครั้ง และฝากเพิ่มทุกต้นปีที่ 2 เป็นต้นไปอีกปีละ 12,000 บาท ให้แสดงตารางยอดเงินแต่ละปีจนกว่ายอดจะถึงเป้าหมาย แล้วสรุปว่าใช้กี่ปี ฝากเงินเองรวมเท่าไร และได้ดอกเบี้ยรวมเท่าไร

สิ่งที่ต้องทำ
  1. ใช้ while เพราะไม่รู้จำนวนปีล่วงหน้า
  2. ลำดับในแต่ละปี: (ถ้าปี ≥ 2) ฝากเพิ่ม → คิดดอกเบี้ย → แสดงยอดสิ้นปี

  4. จัดตารางด้วย printf และ %,.2f
