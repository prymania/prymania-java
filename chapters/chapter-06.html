บทที่ 6 · ให้โปรแกรมทำงานซ้ำ

การทำซ้ำด้วยลูป

ลดคำสั่งที่ต้องเขียนซ้ำ โดยกำหนดว่าจะทำซ้ำเมื่อใด ต้องปรับค่าอะไร และจะออกจากลูปอย่างไร

เมื่อจบบทนี้ นิสิตจะทำได้
  • เขียน while และ for ได้ครบองค์ประกอบ
  • เลือก do-while เมื่อต้องทำอย่างน้อยหนึ่งครั้ง
  • ใช้ลูปซ้อนกับรูปแบบข้อมูล
  • ควบคุมและป้องกันลูปไม่รู้จบ

6. ทำซ้ำโดยไม่เขียนคำสั่งเดิมหลายชุด

สมมติต้องพิมพ์คะแนนของนักศึกษา 100 คน ถ้าไม่มีลูปเราต้องเขียนคำสั่งซ้ำ 100 ชุด ซึ่งยาว แก้ยาก และพลาดได้ง่าย ลูปช่วยให้เขียนขั้นตอนครั้งเดียวแล้วให้โปรแกรมทำซ้ำตามเงื่อนไข

องค์ประกอบ 4 อย่างของลูปที่ดี
  1. ค่าเริ่มต้น (initialization) — ตัวนับหรือตัวแปรควบคุมเริ่มที่ค่าใด
  2. เงื่อนไข (condition) — ทำต่อขณะที่อะไรเป็นจริง
  3. งานที่ทำซ้ำ (body) — สิ่งที่ทำในแต่ละรอบ
  4. การอัปเดต (update) — เปลี่ยนค่าที่ทำให้เงื่อนไขเข้าใกล้ false ถ้าขาดข้อนี้จะเกิดลูปไม่รู้จบ

6.1 while: ทำซ้ำขณะเงื่อนไขเป็นจริง

while เหมาะเมื่อยังไม่รู้จำนวนรอบแน่นอน แต่รู้เงื่อนไขที่ต้องทำต่อ เช่น “ทำต่อขณะที่ยังมีข้อมูล” ก่อนเข้ารอบทุกครั้ง Java ตรวจนิพจน์ในวงเล็บ ซึ่งต้องได้ค่า boolean

ลำดับการทำงานของ while

while (condition) {
    // คำสั่งที่ทำซ้ำ
    // เปลี่ยนค่าที่มีผลต่อ condition
}

  1. ตรวจ condition ก่อนเข้ารอบ
  2. ถ้าเป็น true ทำคำสั่งในวงเล็บปีกกา
  3. จบรอบแล้วกลับไปตรวจ condition ใหม่
  4. ถ้าเป็น false ข้าม body ไปทำคำสั่งถัดจากลูป

ถ้า condition เป็น false ตั้งแต่ครั้งแรก body จะไม่ทำงานเลย (zero iterations)

ตัวอย่าง 6.1.1 · นับ 1 ถึง 5 พื้นฐาน

แนวคิด: ตัวนับเริ่มที่ 1 เพิ่มทีละ 1 และหยุดเมื่อเกิน 5

CountUp.java
public class CountUp {
    public static void main(String[] args) {
        int number = 1;        // ค่าเริ่มต้นของตัวนับ
        while (number <= 5) {  // ตรวจเงื่อนไขก่อนเข้าทุกรอบ
            System.out.println(number);
            number++;  // อัปเดต: ขาดบรรทัดนี้จะวนไม่รู้จบ
        }
        // หลังออกจากลูป number = 6
        System.out.println("After loop: number = " + number);
    }
}
Output
1
2
3
4
5
After loop: number = 6
อธิบายทีละขั้น
  1. ตรวจ 1 ≤ 5 → พิมพ์ 1, number = 2
  2. ทำต่อจนพิมพ์ 5 แล้ว number = 6
  3. ตรวจ 6 ≤ 5 → false ออกจากลูป
  4. หลังลูป number มีค่า 6 (ไม่ใช่ 5)

ตัวอย่าง 6.1.2 · นับถอยหลังด้วย while (i > 0) พื้นฐาน

แนวคิด: เงื่อนไขต้องสอดคล้องกับทิศทางการอัปเดต: ลดค่า → ใช้ > หรือ >=

WhilePositive.java
public class WhilePositive {
    public static void main(String[] args) {
        int i = 3;
        while (i > 0) {  // ลดค่าลงจึงใช้ > 0
            System.out.println(i);
            i--;  // ลดทีละ 1 จนเงื่อนไขเป็นเท็จ
        }
        System.out.println("Done");
    }
}
Output
3
2
1
Done
อธิบายทีละขั้น
  1. 3 > 0 → พิมพ์ 3, i = 2
  2. 2 > 0 → พิมพ์ 2, i = 1
  3. 1 > 0 → พิมพ์ 1, i = 0
  4. 0 > 0 false → ออก พิมพ์ Done

ตัวอย่าง 6.1.3 · สะสมผลรวม (accumulator) ต่อยอด

แนวคิด: รูปแบบสำคัญที่สุดของลูป: ตัวแปรผลรวมเริ่มที่ 0 แล้วบวกค่าของแต่ละรอบเข้าไป

SumToN.java
public class SumToN {
    public static void main(String[] args) {
        int n = 5;
        int number = 1;
        int sum = 0;  // ตัวสะสมเริ่มที่ 0
        while (number <= n) {
            sum += number;  // บวกค่าของรอบนี้เข้าไป
            System.out.println("add " + number + " -> sum = " + sum);
            number++;
        }
        System.out.println("1 + ... + " + n + " = " + sum);
    }
}
Output
add 1 -> sum = 1
add 2 -> sum = 3
add 3 -> sum = 6
add 4 -> sum = 10
add 5 -> sum = 15
1 + ... + 5 = 15
อธิบายทีละขั้น
  1. sum เริ่ม 0 (ถ้าเริ่มค่าอื่นผลรวมจะผิด)
  2. แต่ละรอบ sum += number
  3. พิมพ์ระหว่างทางช่วยให้เห็นการเปลี่ยนแปลง (เทคนิค debug)
  4. ผลรวม 15 ตรงกับสูตร n(n+1)/2

ตัวอย่าง 6.1.4 · ไม่รู้จำนวนรอบ: นับหลักของตัวเลข ประยุกต์

แนวคิด: จำนวนรอบขึ้นกับข้อมูล — หาร 10 ไปเรื่อย ๆ จนเหลือ 0 นี่คือจุดที่ while เหมาะกว่า for

CountDigits.java
import java.util.Scanner;

public class CountDigits {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Number: ");
        int n = input.nextInt();
        int digits = 0;
        int digitSum = 0;
        int temp = n;  // ใช้สำเนา เพื่อไม่ให้ค่า n เปลี่ยน
        while (temp > 0) {
            digitSum += temp % 10;  // % 10 = หลักขวาสุด
            temp /= 10;             // ตัดหลักขวาสุดทิ้ง
            digits++;               // นับจำนวนหลัก
        }
        System.out.println(n + " has " + digits + " digits, digit sum = " + digitSum);
    }
}
Output (ตัวที่ขีดเส้นใต้ = ค่าที่ผู้ใช้พิมพ์)
Number: 907215
907215 has 6 digits, digit sum = 24
อธิบายทีละขั้น
  1. temp = 907215 → หลักหน่วย 5, temp = 90721
  2. ทำซ้ำ: 1, 2, 7, 0, 9
  3. temp เป็น 0 → หยุด นับได้ 6 หลัก ผลรวม 24
  4. ใช้ temp แทน n เพื่อเก็บค่าเดิมไว้แสดงผล

ตัวอย่าง 6.1.5 · while (true) กับ break ประยุกต์

แนวคิด: บางครั้งเขียนลูปที่ตั้งใจไม่หยุดเอง แล้วใช้ break ออกเมื่อถึงเงื่อนไข — ต้องแน่ใจว่า break ไปถึงได้จริง

WhileTrueBreak.java
public class WhileTrueBreak {
    public static void main(String[] args) {
        int attempt = 1;
        while (true) {  // เงื่อนไขจริงเสมอ ต้องมี break
            System.out.println("Attempt " + attempt);
            if (attempt >= 3) {
                break;  // ออกจากลูปทันที
            }
            attempt++;
        }
        System.out.println("Stopped safely");
    }
}
Output
Attempt 1
Attempt 2
Attempt 3
Stopped safely
อธิบายทีละขั้น
  1. ลูปพิมพ์ Attempt 1, 2, 3
  2. เมื่อ attempt = 3 เงื่อนไข if จริง → break
  3. ไปทำคำสั่งหลังลูป

ทดลอง error 6.1.6 · assignment ไม่ใช่เงื่อนไข ประยุกต์

แนวคิด: while (i = 3) เป็นการกำหนดค่า ได้ชนิด int ไม่ใช่ boolean

public class WhileConditionError {

            System.out.println(i);
            i--;
        }

  1. compiler แจ้ง int cannot be converted to boolean
  2. ต้องการเปรียบเทียบใช้ == แต่ในกรณีนี้ควรเป็น i > 0

while (number <= 5) {

ลองคิดก่อน · นับรอบ
int x = 20;
int rounds = 0;
while (x > 1) {

6.2 for: เมื่อรู้จำนวนรอบหรือตัวนับชัดเจน

for รวมจุดเริ่ม เงื่อนไข และการอัปเดตไว้ในบรรทัดเดียว ทำให้อ่านแล้วรู้จำนวนรอบทันที

Syntax และ flow ของ for

for (initialization; condition; update) {
    // loop body
}

  1. ทำ initialization ครั้งเดียวก่อนเริ่มลูป
  2. ตรวจ condition; ถ้า false ออกจากลูป
  3. ถ้า true ทำ body
  4. ทำ update แล้วกลับไปข้อ 2

ตัวแปรที่ประกาศในส่วนเริ่มต้น เช่น int i ใช้ได้เฉพาะภายในลูปนั้น

ตัวอย่าง 6.2.1 · อ่าน header ของ for ตามลำดับ พื้นฐาน

แนวคิด: ก่อนรัน ให้ทำนายค่าเริ่มต้น จำนวนรอบ และผลรวม

ForFlow.java
public class ForFlow {
    public static void main(String[] args) {
        int sum = 0;
        // ค่าเริ่ม; เงื่อนไข; อัปเดต
        for (int i = 1; i <= 3; i++) {
            sum += i;
            System.out.println("i=" + i + ", sum=" + sum);
        }
        System.out.println("After loop, sum=" + sum);
    }
}
Output
i=1, sum=1
i=2, sum=3
i=3, sum=6
After loop, sum=6
อธิบายทีละขั้น
  1. i = 1 → ตรวจ 1 ≤ 3 → body → i++
  2. i = 2, 3 ทำเช่นเดียวกัน
  3. i = 4 → เงื่อนไข false → ออก
  4. i ใช้นอกลูปไม่ได้ เพราะประกาศใน header

ตัวอย่าง 6.2.2 · ขยับทีละมากกว่า 1 และนับถอยหลัง พื้นฐาน

ตัวอย่าง 6.2.4 · หาค่ามาก/น้อยสุดและค่าเฉลี่ยจากข้อมูล n ค่า ประยุกต์

แนวคิด: รูปแบบ “ติดตามค่าที่ดีที่สุด”: กำหนดค่าเริ่มจากข้อมูลตัวแรก แล้วเทียบกับตัวถัดไปทีละตัว

MinMaxAverage.java
import java.util.Scanner;

public class MinMaxAverage {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("How many scores? ");
        int n = input.nextInt();
        System.out.print("Scores: ");
        // ค่าแรกเป็นทั้ง max, min และผลรวมเริ่มต้น
        int first = input.nextInt();
        int max = first, min = first, sum = first;
        // เริ่มรอบที่ 2 เพราะอ่านค่าแรกไปแล้ว
        for (int i = 2; i <= n; i++) {
            int score = input.nextInt();
            sum += score;
            if (score > max) max = score;
            if (score < min) min = score;
        }
        // แปลงเป็น double ก่อนหาร
        System.out.printf("Max=%d Min=%d Avg=%.2f%n", max, min, (double) sum / n);
    }
}

  3. แต่ละรอบปรับ max/min ถ้าเจอค่าที่ดีกว่า

เลือก for หรือ while

ใช้ for เมื่อรู้ช่วงหรือจำนวนรอบชัดเจน (นับ 1–100, วนตามจำนวนข้อมูล) ใช้ while เมื่อทำซ้ำจนกว่าเหตุการณ์หนึ่งจะเกิด (ผู้ใช้ป้อน 0, ตัวเลขหารจนเหลือ 0) ทั้งคู่ทำงานแทนกันได้ แต่รูปแบบที่ตรงโจทย์อ่านง่ายกว่า

6.3 do-while และการทำซ้ำด้วย sentinel

do-while ตรวจเงื่อนไขหลังทำ body จึงทำอย่างน้อยหนึ่งครั้ง เหมาะกับการแสดงเมนูหรือถามข้อมูลก่อน แล้วค่อยตรวจว่าต้องทำต่อหรือไม่ ส่วน sentinel คือค่าพิเศษที่ใช้สั่งหยุด เช่น ป้อน 0 หรือ -1

Syntax และ flow ของ do-while

do {
    // คำสั่งที่ต้องทำอย่างน้อยหนึ่งครั้ง
} while (condition);   // ← มี ; ปิดท้าย

  1. เข้าทำ body ทันทีโดยยังไม่ตรวจเงื่อนไข
  2. ทำคำสั่งใน body และปรับค่าที่เกี่ยวข้อง
  3. ตรวจ condition ที่ท้ายลูป
  4. ถ้า true กลับไปเริ่ม body; ถ้า false ออกจากลูป

ตัวอย่าง 6.3.1 · ทำอย่างน้อยหนึ่งครั้ง พื้นฐาน

แนวคิด: แม้เงื่อนไขเป็น false ตั้งแต่แรก body ก็ทำไปแล้วหนึ่งครั้ง — ต่างจาก while

DoWhileOnce.java
public class DoWhileOnce {
    public static void main(String[] args) {
        int count = 10;
        do {  // ทำ body ก่อน แล้วค่อยตรวจเงื่อนไข
            System.out.println("do-while body, count = " + count);
            count++;
        } while (count <= 3);  // มี ; ปิดท้าย do-while

        int k = 10;
        while (k <= 3) {  // while ตรวจก่อน → ไม่เข้าเลย
            System.out.println("while body");
            k++;
        }
        System.out.println("Finished");
    }
}
Output
do-while body, count = 10
Finished
อธิบายทีละขั้น
  1. do-while ทำ body ก่อน (count = 10) แล้วตรวจ 11 ≤ 3 → false จบ
  2. while ตรวจ 10 ≤ 3 ก่อน → false ไม่ทำเลย
  3. ผลจึงต่างกัน 1 รอบ

ตัวอย่าง 6.3.2 · บังคับป้อนค่าในช่วงที่ถูกต้อง (input validation) ต่อยอด

แนวคิด: ต้องถามอย่างน้อยหนึ่งครั้งเสมอ แล้วถามซ้ำถ้าค่าไม่ถูกต้อง → do-while พอดี

import java.util.Scanner;

public class ValidScore {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int score;

                System.out.println("  Invalid, try again.");
            }
        // ค่าผิดช่วง → วนกลับไปถามใหม่
        } while (score < 0 || score > 100);
        System.out.println("Accepted: " + score);
    }
}

  2. -5 นอกช่วง → วนใหม่
  3. 87 ในช่วง → เงื่อนไขท้าย false → ออก

ตัวอย่าง 6.3.3 · รวมค่าจนป้อน sentinel 0 ประยุกต์

แนวคิด: อ่านค่าก่อนตรวจ (priming read) แล้วอ่านค่าถัดไปที่ท้าย body — sentinel จะไม่ถูกบวกเข้าไป

SentinelSum.java
import java.util.Scanner;

public class SentinelSum {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int sum = 0, count = 0;  // ตัวสะสมและตัวนับ
        System.out.print("Number (0 to stop): ");
        int number = input.nextInt();  // รับค่าแรกก่อนเข้าลูป
        // 0 คือค่าหยุด (sentinel)
        while (number != 0) {
            sum += number;
            count++;
            System.out.print("Number (0 to stop): ");
            // รับค่าใหม่ก่อนตรวจเงื่อนไขรอบถัดไป
            number = input.nextInt();
        }
        System.out.println("Count = " + count + ", Sum = " + sum);
        if (count > 0) {  // ป้องกันการหารด้วย 0
            System.out.printf("Average = %.2f%n", (double) sum / count);
        }
    }
}

Number (0 to stop): 0

  2. บวก 12, 7, 30 ตามลำดับ
  3. อ่าน 0 → เงื่อนไข false ออกก่อนบวก
  4. ตรวจ count > 0 ก่อนหาร ป้องกันการหารด้วยศูนย์

ตัวอย่าง 6.3.4 · เมนูวนซ้ำด้วย do-while + switch ท้าทาย

แนวคิด: โครงสร้างพื้นฐานของโปรแกรมเมนู: แสดงเมนู → รับตัวเลือก → ทำตามตัวเลือก → วนจนกว่าจะเลือกออก

import java.util.Scanner;

public class BankMenu {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        double balance = 1000;
        int choice;
        do {  // แสดงเมนูอย่างน้อย 1 ครั้ง
            System.out.println("[1] Deposit [2] Withdraw [3] Balance [0] Exit");
            System.out.print("Choice: ");
            choice = input.nextInt();
            switch (choice) {  // เลือกงานตามเมนู

                    System.out.print("Amount: ");
                    balance += input.nextDouble();

                case 3 -> System.out.printf("Balance: %.2f%n", balance);
                case 0 -> System.out.println("Goodbye");
                default -> System.out.println("Invalid choice");
            }
        } while (choice != 0);  // เลือก 0 จึงออก
    }
}

public class PrimesUpTo50 {
    public static void main(String[] args) {
        int count = 0;
        for (int n = 2; n <= 50; n++) {
            // สมมติว่าเป็นจำนวนเฉพาะก่อน
            boolean isPrime = true;
            // ตรวจตัวหารถึงรากที่สองของ n พอ

                    break;  // ออกเฉพาะลูป d
                }
            }
            if (isPrime) {
                System.out.print(n + " ");
                count++;
            }
        }
        System.out.println();
        System.out.println("Total primes: " + count);
    }
}

import java.util.Scanner;

public class ValidateAndGrade {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int score;
        int wrong = 0;
        do {
            System.out.print("Score: ");
            score = input.nextInt();
            // ค่าผิดช่วง → นับจำนวนครั้งที่ผิด
            if (score < 0 || score > 100) {
                System.out.println("Out of range!");
                wrong++;
            }
        // วนจนกว่าจะได้ค่าในช่วง
        } while (score < 0 || score > 100);
        char grade;
        if (score >= 80) grade = 'A';
        else if (score >= 70) grade = 'B';
        else if (score >= 60) grade = 'C';
        else if (score >= 50) grade = 'D';
        else grade = 'F';
        System.out.println("Grade " + grade + " (invalid attempts: " + wrong + ")");
    }
}

import java.util.Scanner;

public class ScoreStats {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int count = 0, sum = 0, passed = 0;
        // ค่าเริ่มที่ทำให้ค่าแรกชนะเสมอ
        int max = Integer.MIN_VALUE, min = Integer.MAX_VALUE;
        System.out.print("Scores (-1 to stop): ");
        int score = input.nextInt();
        while (score != -1) {  // -1 คือค่าหยุด
            count++;
            sum += score;
            if (score > max) max = score;
            if (score < min) min = score;
            if (score >= 50) passed++;
            // อ่านค่าถัดไปก่อนตรวจเงื่อนไข
            score = input.nextInt();
        }
        if (count == 0) {  // ป้องกันการหารด้วย 0
            System.out.println("No data");
        } else {
            System.out.println("Count: " + count);
            System.out.println("Max: " + max + ", Min: " + min);
            System.out.printf("Average: %.2f%n", (double) sum / count);
            System.out.println("Passed: " + passed);
        }
    }
}

import java.util.Scanner;

public class ReverseNumber {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Number: ");
        int n = input.nextInt();
        int temp = n, reversed = 0;
        while (temp > 0) {
            // เลื่อนหลักเดิมไปซ้าย แล้วเติมหลักขวาสุดของ temp

import java.util.Scanner;

public class GuessNumber {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int secret = 37;
        boolean found = false;
        // ทายได้ไม่เกิน 6 ครั้ง
        for (int attempt = 1; attempt <= 6; attempt++) {
            System.out.print("Guess #" + attempt + ": ");
            int guess = input.nextInt();
            if (guess == secret) {
                System.out.println("Correct! You used " + attempt + " guesses.");
                found = true;
                break;  // ทายถูก → ออกจากลูป

                System.out.println("Too high");
            } else {
                System.out.println("Too low");

import java.util.Scanner;

public class NumberDiamond {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("h = ");
        int h = input.nextInt();
        // ครึ่งบน: แถวยาวขึ้น
        for (int k = 1; k <= h; k++) printRow(k, h);
        // ครึ่งล่าง: แถวสั้นลง
        for (int k = h - 1; k >= 1; k--) printRow(k, h);
    }

    // เมธอดพิมพ์ 1 แถว (บทที่ 7)
    static void printRow(int k, int h) {
        // เว้นวรรคด้านหน้า
        for (int s = 1; s <= h - k; s++) System.out.print(" ");
        // นับขึ้น 1..k
        for (int i = 1; i <= k; i++) System.out.print(i);
        // นับลง k-1..1
        for (int i = k - 1; i >= 1; i--) System.out.print(i);
        System.out.println();
    }
}

Exercise 6.11จำนวนสมบูรณ์และจำนวนเฉพาะในช่วง★★★ ท้าทาย

รับช่วง a ถึง b แล้วแสดง (1) จำนวนเฉพาะทั้งหมดในช่วง (2) จำนวนสมบูรณ์ (perfect number: ผลรวมตัวหารแท้เท่ากับตัวมันเอง เช่น 6 = 1 + 2 + 3) ทั้งหมดในช่วง และ (3) จำนวนของแต่ละประเภท

สิ่งที่ต้องทำ
  1. ลูปนอกวน n จาก a ถึง b
  2. ตรวจจำนวนเฉพาะด้วยลูปตัวหารถึง √n และ break เมื่อเจอตัวหาร
  3. หาผลรวมตัวหารแท้ด้วยลูปอีกชุด
  4. จำนวนที่น้อยกว่า 2 ไม่ใช่จำนวนเฉพาะ

PrimeAndPerfect.java
import java.util.Scanner;

public class PrimeAndPerfect {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Range a b: ");
        int a = input.nextInt(), b = input.nextInt();
        int primeCount = 0, perfectCount = 0;
        System.out.print("Primes: ");
        for (int n = a; n <= b; n++) {
            if (n < 2) continue;  // 0 และ 1 ไม่ใช่จำนวนเฉพาะ
            boolean prime = true;
            // ตรวจตัวหารถึงรากที่สองของ n
            for (int d = 2; d * d <= n; d++) {
                if (n % d == 0) { prime = false; break; }
            }
            if (prime) { System.out.print(n + " "); primeCount++; }
        }
        System.out.println("(" + primeCount + ")");
        System.out.print("Perfect: ");
        for (int n = Math.max(a, 2); n <= b; n++) {
            int sum = 0;
            // ผลรวมตัวหารที่น้อยกว่า n
            for (int d = 1; d < n; d++) {
                if (n % d == 0) sum += d;
            }
            // ผลรวมตัวหารเท่ากับตัวเอง = จำนวนสมบูรณ์
            if (sum == n) { System.out.print(n + " "); perfectCount++; }
        }
        System.out.println("(" + perfectCount + ")");
    }
}

CompoundSavings.java
import java.util.Scanner;

public class CompoundSavings {
    public static void main(String[] args) {
        final double YEARLY_DEPOSIT = 12000;
        Scanner input = new Scanner(System.in);
        System.out.print("Principal: ");
        double balance = input.nextDouble();
        System.out.print("Rate (% per year): ");
        double rate = input.nextDouble();
        System.out.print("Goal: ");
        double goal = input.nextDouble();

        double deposited = balance;
        int year = 0;
        System.out.printf("%4s %12s %12s%n", "Year", "Interest", "Balance");
        while (balance < goal) {  // ทำซ้ำจนกว่าจะถึงเป้าหมาย
            year++;
            if (year >= 2) {  // ฝากเพิ่มตั้งแต่ปีที่ 2
                balance += YEARLY_DEPOSIT;
                deposited += YEARLY_DEPOSIT;
            }
            // ดอกเบี้ยทบต้นจากยอดปัจจุบัน
            double interest = balance * rate / 100;
            balance += interest;
            System.out.printf("%4d %,12.2f %,12.2f%n", year, interest, balance);
        }
        System.out.println("Years needed: " + year);
        System.out.printf("You deposited: %,.2f%n", deposited);
        System.out.printf("Total interest: %,.2f%n", balance - deposited);
    }
}
