บทที่ 11 · เริ่มต้นการเขียนโปรแกรมเชิงวัตถุ

คลาสและออบเจ็กต์

รวมข้อมูลและพฤติกรรมที่เกี่ยวข้องไว้ด้วยกันเป็นคลาส แล้วสร้างออบเจ็กต์หลายตัวที่มีสถานะของตัวเอง

เมื่อจบบทนี้ นิสิตจะทำได้
  • อธิบายความต่างของ class กับ object
  • สร้าง object ด้วย new และเรียก method
  • เขียน constructor และใช้ this
  • จัดการหลาย object ที่มี state แยกกัน

11. จากตัวแปรหลายตัว สู่ออบเจ็กต์เดียว

ถ้าต้องเก็บข้อมูลนักศึกษา 3 คน คนละ ชื่อ รหัส คะแนน เราอาจใช้อาเรย์คู่ขนาน 3 ชุด (บทที่ 8) ซึ่งต้องระวังให้ index ตรงกันเสมอ การเขียนโปรแกรมเชิงวัตถุ (OOP) ให้เรารวมข้อมูลที่เกี่ยวข้องเป็นก้อนเดียว พร้อมเมธอดที่ทำงานกับข้อมูลนั้น

คำศัพท์หลัก
  • class — พิมพ์เขียว บอกว่าวัตถุชนิดนี้มีข้อมูล (field) และพฤติกรรม (method) อะไร
  • object / instance — ของจริงที่สร้างจากพิมพ์เขียว แต่ละตัวมีค่า field ของตัวเอง
  • field (attribute) — ตัวแปรที่อยู่ในออบเจ็กต์ เช่น name, balance
  • method — สิ่งที่ออบเจ็กต์ทำได้ เช่น deposit(), describe()
  • state — ค่าของ field ทั้งหมด ณ ขณะหนึ่ง

11.1 class เป็นพิมพ์เขียว

class เป็นพิมพ์เขียวที่บอกว่าของชนิดหนึ่งมีข้อมูลอะไร (field) และทำอะไรได้ (method) ส่วน object คือของจริงที่สร้างจากพิมพ์เขียวนั้น สร้างได้หลายตัว แต่ละตัวมีค่า field ของตัวเอง

คลาส (พิมพ์เขียว)field (มีอะไร)method (ทำอะไร)object (ของจริง)
Booktitle, author, pagesdescribe(), isThick()หนังสือ “Java Basics” 320 หน้า
BankAccountowner, balancedeposit(), withdraw()บัญชีของ JHope ยอด 1000
Studentname, scoreisPassed()Suga คะแนน 72
โครงสร้างของคลาส
Student.java
class Student {               // ชื่อคลาส: PascalCase
    String name;              // field (ข้อมูล)
    int score;

    void describe() {         // method (พฤติกรรม) — ไม่มี static
        System.out.println(name + ": " + score);
    }
}

ตัวอย่างในบทนี้วางหลายคลาสไว้ในไฟล์เดียว คลาสที่มี main เป็น public และคลาสอื่นไม่ใส่ public ส่วนในโปรเจกต์จริงนิยมแยกไฟล์ละคลาส

ตัวอย่าง 11.1.1คลาสแรก: Book

BookDemo.java
public class BookDemo {
    public static void main(String[] args) {
        Book b = new Book();                  // สร้างออบเจ็กต์จากคลาส Book
        b.title = "Java Basics";              // เข้าถึง field ด้วยจุด (.)
        b.author = "JHope";
        b.pages = 320;
        System.out.println(b.title + " by " + b.author + ", " + b.pages + " pages");
    }
}

class Book {                                  // พิมพ์เขียว: ทุก Book มี 3 field นี้
    String title;
    String author;
    int pages;
}

Java Basics by JHope, 320 pages

เพิ่มพฤติกรรมด้วย method

เมธอดในคลาสไม่มี static เพราะเป็นของแต่ละออบเจ็กต์ และใช้ field ของออบเจ็กต์ที่ถูกเรียกได้ทันที ไม่ต้องส่งเป็น parameter

ตัวอย่าง 11.1.2เมธอดใช้ field ของตัวเอง

BookMethods.java
public class BookMethods {
    public static void main(String[] args) {
        Book b = new Book();
        b.title = "Java Basics";
        b.pages = 320;
        b.describe();                                                    // Book: Java Basics (320 pages)
        System.out.println("Reading time: " + b.readingHours(40) + " hours");  // 320 / 40 = 8.0
        System.out.println("Thick book? " + b.isThick());                // true
    }
}

class Book {
    String title;
    int pages;

    void describe() {                         // ใช้ title และ pages ของ b
        System.out.println("Book: " + title + " (" + pages + " pages)");
    }

    double readingHours(int pagesPerHour) {   // รับ parameter และคืนค่าได้ตามปกติ
        return (double) pages / pagesPerHour;
    }

    boolean isThick() {
        return pages > 300;
    }
}
ผลลัพธ์
Book: Java Basics (320 pages)
Reading time: 8.0 hours
Thick book? true

ลองคิดก่อน · class vs object

ในประโยค “แมวชื่อส้มของฉัน อายุ 3 ปี” อะไรคือ class อะไรคือ object และอะไรคือ field

11.2 สร้าง object ด้วย new และตัวแปรอ้างอิง

new Product() สร้างออบเจ็กต์ในหน่วยความจำ ตัวแปร Product p ไม่ได้เก็บออบเจ็กต์ไว้ข้างใน แต่เก็บการอ้างอิง (reference) ที่ชี้ไปหาออบเจ็กต์ เหมือนอาเรย์ในบทที่ 8

field ที่ยังไม่กำหนดค่าได้ค่าเริ่มต้นอัตโนมัติ ต่างจากตัวแปร local ในเมธอดที่ต้องกำหนดค่าก่อนใช้

ตัวอย่าง 11.2.1ค่าเริ่มต้นของ field

DefaultFields.java
public class DefaultFields {
    public static void main(String[] args) {
        Product p = new Product();                    // ยังไม่ได้กำหนด field ใดเลย
        System.out.println("name = " + p.name);       // null
        System.out.println("price = " + p.price);     // 0.0
        System.out.println("stock = " + p.stock);     // 0
        System.out.println("onSale = " + p.onSale);   // false
    }
}

class Product {
    String name;
    double price;
    int stock;
    boolean onSale;
}
ผลลัพธ์
name = null
price = 0.0
stock = 0
onSale = false

สองตัวแปรชี้ออบเจ็กต์เดียวกัน

b = a คัดลอกการอ้างอิง ไม่ได้สร้างออบเจ็กต์ใหม่ แก้ผ่าน b ก็เห็นผ่าน a ด้วย ต้องใช้ new เท่านั้นจึงได้ออบเจ็กต์ใหม่

ตัวอย่าง 11.2.2คัดลอกการอ้างอิง vs สร้างใหม่

SharedReference.java
public class SharedReference {
    public static void main(String[] args) {
        Product a = new Product();
        a.name = "Pen";
        a.stock = 10;

        Product b = a;                                // b ชี้ออบเจ็กต์เดียวกับ a
        b.stock = 3;                                  // แก้ผ่าน b

        Product c = new Product();                    // ออบเจ็กต์ใหม่อีกตัว
        c.name = "Pen";
        c.stock = 3;

        System.out.println("a.stock = " + a.stock);   // 3   a เห็นค่าที่แก้ผ่าน b
        System.out.println("a == b ? " + (a == b));   // true   ชี้ตัวเดียวกัน
        System.out.println("a == c ? " + (a == c));   // false  ค่าเหมือนแต่คนละตัว
    }
}

class Product {
    String name;
    int stock;
}
ผลลัพธ์
a.stock = 3
a == b ? true
a == c ? false
คำแนะนำ

== กับออบเจ็กต์เทียบว่าชี้ตัวเดียวกันหรือไม่ ไม่ได้เทียบค่าใน field เหมือนที่ต้องใช้ equals กับ String ในบทที่ 9

ส่งออบเจ็กต์ให้เมธอด

เมธอดที่รับออบเจ็กต์ได้สำเนาของการอ้างอิง จึงชี้ออบเจ็กต์ตัวจริงและแก้ field ได้ ต่างจากการส่ง int ในบทที่ 7 ที่แก้ค่าเดิมไม่ได้

ตัวอย่าง 11.2.3เมธอดแก้ field ของออบเจ็กต์ตัวจริง

PassObject.java
public class PassObject {
    public static void main(String[] args) {
        Product p = new Product();
        p.name = "Notebook";
        p.price = 40;
        applyDiscount(p, 25);                         // item กับ p ชี้ตัวเดียวกัน
        System.out.println(p.name + " now costs " + p.price);   // 30.0
    }

    static void applyDiscount(Product item, double percent) {
        item.price = item.price * (100 - percent) / 100;   // 40 × 75 / 100
    }
}

class Product {
    String name;
    double price;
}
ผลลัพธ์
Notebook now costs 30.0

11.3 constructor และ this

การกำหนด field ทีละบรรทัดหลัง new ยาวและลืมง่าย constructor คือเมธอดพิเศษที่ทำงานอัตโนมัติตอน new ใช้กำหนดค่าเริ่มต้นให้ field

กฎของ constructor
ชื่อเหมือนชื่อคลาสทุกตัวอักษร
ชนิดผลลัพธ์ไม่มี (ไม่ใส่แม้แต่ void)
เรียกเมื่อไรอัตโนมัติตอน new พร้อม argument ในวงเล็บ
thisหมายถึง “ออบเจ็กต์ตัวนี้” ใช้แยก field กับ parameter ที่ชื่อซ้ำกัน

ตัวอย่าง 11.3.1constructor กำหนดค่าตั้งแต่สร้าง

ConstructorDemo.java
public class ConstructorDemo {
    public static void main(String[] args) {
        Student a = new Student("JHope", 85);     // ส่ง argument ให้ constructor
        Student b = new Student("Suga", 72);
        a.describe();
        b.describe();
    }
}

class Student {
    String name;
    int score;

    Student(String name, int score) {         // ชื่อเดียวกับคลาส ไม่มีชนิดผลลัพธ์

หลาย constructor และ this(...)

คลาสมี constructor หลายตัวได้ ถ้าจำนวนหรือชนิดของ parameter ต่างกัน (overloading) ตัวที่สั้นกว่าเรียกตัวหลักด้วย this(...) เพื่อไม่ต้องเขียนโค้ดซ้ำ

ตัวอย่าง 11.3.3สี่เหลี่ยมสร้างได้ 3 แบบ

OverloadedConstructors.java
public class OverloadedConstructors {
    public static void main(String[] args) {
        Rectangle a = new Rectangle(4, 3);        // ใช้ constructor 2 ค่า
        Rectangle b = new Rectangle(5);           // จัตุรัส 5 x 5
        Rectangle c = new Rectangle();            // ค่าเริ่มต้น 1 x 1
        System.out.println(a.info());
        System.out.println(b.info());
        System.out.println(c.info());
    }
}

class Rectangle {
    double width;
    double height;

    Rectangle(double width, double height) {  // constructor หลัก
        this.width = width;
        this.height = height;
    }

    Rectangle(double side) {
        this(side, side);                     // เรียก constructor หลัก
    }

    Rectangle() {
        this(1);                              // → this(1, 1)
    }

    double area() {
        return width * height;
    }

    String info() {
        return width + " x " + height + " = " + area();   // เรียกเมธอดในคลาสเดียวกันได้ตรง ๆ
    }
}
ผลลัพธ์

5.0 x 5.0 = 25.0
1.0 x 1.0 = 1.0

ToStringDemo.java
public class ToStringDemo {
    public static void main(String[] args) {
        Point p = new Point(3, 4);
        Point q = new Point(-1, 2);
        System.out.println(p);                        // เรียก p.toString() → (3, 4)
        System.out.println("q is " + q);              // ต่อ String ก็เรียก toString()
        System.out.printf("distance = %.2f%n", p.distanceTo(q));
    }
}

class Point {
    int x;
    int y;

    Point(int x, int y) {
        this.x = x;
        this.y = y;
    }

    double distanceTo(Point other) {              // รับออบเจ็กต์ชนิดเดียวกัน
        int dx = x - other.x;                     // x ของตัวเอง − x ของ other = 4
        int dy = y - other.y;                     // 2
        return Math.sqrt(dx * dx + dy * dy);      // √20
    }

    @Override                                     // ตั้งใจเขียนทับ toString ที่มีอยู่แล้ว (บทที่ 12)
    public String toString() {
        return "(" + x + ", " + y + ")";
    }
}

TwoAccounts.java
public class TwoAccounts {
    public static void main(String[] args) {
        BankAccount jhope = new BankAccount("JHope", 1000);
        BankAccount suga = new BankAccount("Suga", 500);
        jhope.deposit(250);                       // jhope: 1250
        suga.withdraw(200);                       // suga: 300   jhope ไม่เปลี่ยน
        suga.withdraw(1000);                      // เกินยอด → ปฏิเสธ
        jhope.printBalance();
        suga.printBalance();
    }
}

class BankAccount {
    String owner;
    double balance;

    BankAccount(String owner, double balance) {
        this.owner = owner;
        this.balance = balance;
    }

    void deposit(double amount) {
        balance += amount;
    }

    void withdraw(double amount) {
        if (amount > balance) {                   // กฎอยู่ในคลาส main ไม่ต้องตรวจเอง
            System.out.println(owner + ": insufficient funds for " + amount);
        } else {
            balance -= amount;
        }
    }

    void printBalance() {
        System.out.printf("%s: %.2f%n", owner, balance);
    }
}

public class StudentArray {
    public static void main(String[] args) {
        Student[] list = {                        // อาเรย์ 4 ช่อง แต่ละช่องชี้ Student หนึ่งคน
            new Student("JHope", 85),
            new Student("Suga", 62),
            new Student("RM", 91),
            new Student("Jungkook", 48)
        };
        Student best = list[0];                   // เริ่มจากคนแรก
        int passed = 0;
        for (Student s : list) {                  // s = ทีละคน
            System.out.println(s);                // เรียก toString()
            if (s.score > best.score) best = s;   // จำคนที่คะแนนสูงกว่า
            if (s.isPassed()) passed++;
        }
        System.out.println("Top: " + best.name);  // RM
        System.out.println("Passed: " + passed + "/" + list.length);   // 3/4
    }
}

class Student {
    String name;
    int score;

    Student(String name, int score) {
        this.name = name;
        this.score = score;
    }

    boolean isPassed() {
        return score >= 50;
    }

    public String toString() {
        return String.format("%-8s %3d %s", name, score, isPassed() ? "PASS" : "FAIL");
    }
}
