บทที่ 7 · จัดกลุ่มคำสั่งให้เรียกใช้ซ้ำ

เมธอด: แบ่งโปรแกรมเป็นส่วน

เมธอดช่วยตั้งชื่อให้การทำงานหนึ่งอย่าง เรียกใช้ซ้ำได้ และทำให้ main ไม่ต้องรับผิดชอบทุกขั้นตอนเอง

เมื่อจบบทนี้ นิสิตจะทำได้
  • ประกาศและเรียกใช้ static method
  • ส่งข้อมูลผ่าน parameter
  • คืนผลลัพธ์ด้วย return
  • อธิบายขอบเขตของตัวแปรและการ overload

7. ย้ายงานย่อยไปไว้ในเมธอด

เมธอด (method) คือชุดคำสั่งที่ตั้งชื่อไว้เพื่อทำงานหนึ่งอย่าง เช่น คำนวณราคา หรือพิมพ์ใบเสร็จ เมื่อโปรแกรมยาวขึ้น เราแยกงานเป็นเมธอดเล็ก ๆ เพื่อให้อ่านง่าย เรียกซ้ำได้ และตรวจข้อผิดพลาดทีละส่วนได้ จริง ๆ แล้วเราใช้เมธอดมาตั้งแต่บทแรก: main, println, nextInt, Math.sqrt ล้วนเป็นเมธอด บทนี้จะเขียนเมธอดของเราเอง

7.1 ประกาศและเรียกใช้เมธอด

เมธอดคือชุดคำสั่งที่ตั้งชื่อไว้ทำงานหนึ่งอย่าง ประกาศในคลาส แต่นอกเมธอดอื่น (ไม่ซ้อนใน main) แล้วเรียกด้วยชื่อตามด้วย () กี่ครั้งก็ได้

ส่วนความหมาย
staticเรียกได้โดยไม่ต้องสร้าง object (ความหมายเต็มอยู่ในบทที่ 11–12)
voidชนิดผลลัพธ์: ไม่ส่งค่ากลับ
printWelcomeชื่อเมธอด lowerCamelCase ขึ้นต้นด้วยคำกริยา
( )parameter (ตอนนี้ยังว่าง)
{ ... }body: คำสั่งที่ทำเมื่อถูกเรียก

ตัวอย่าง 7.1.1แยกการแสดงข้อความออกจาก main

Welcome.java
public class Welcome {
    public static void main(String[] args) {
        printWelcome();                           // กระโดดไปทำ แล้วกลับมาบรรทัดถัดไป
        System.out.println("(main continues)");
        printWelcome();                           // เรียกซ้ำได้
    }

    static void printWelcome() {
        System.out.println("Welcome to Java");
    }
}
ผลลัพธ์
Welcome to Java
(main continues)
Welcome to Java

เมธอดเรียกเมธอดอื่นต่อได้

โปรแกรมจำไว้เสมอว่าต้องกลับไปที่ใด เมื่อเมธอดทำงานจบจะกลับไปทำต่อที่บรรทัดถัดจากจุดที่เรียก

ตัวอย่าง 7.1.2ไล่ลำดับการเรียก

CallChain.java
public class CallChain {
    public static void main(String[] args) {
        System.out.println("main: start");      // 1
        printHeader();                          // 2 → ไปที่ printHeader
        System.out.println("main: end");        // 6
    }

    static void printHeader() {
        printLine();                            // 3 → ไปที่ printLine
        System.out.println("  REPORT");         // 4
        printLine();                            // 5
    }

    static void printLine() {
        System.out.println("==========");
    }
}
ผลลัพธ์
main: start
==========
  REPORT
==========
main: end

void ไม่ได้แปลว่าไม่มี output

เมธอด void พิมพ์ข้อความได้ตามปกติ “void” หมายถึงไม่ส่งค่ากลับให้ผู้เรียกนำไปใช้ต่อ การแสดงผล (print) กับการคืนค่า (return) เป็นคนละเรื่อง จะเห็นชัดในหัวข้อ 7.3

ลองคิดก่อน · ลำดับการเรียก
Java
static void a() { System.out.print("A"); b(); System.out.print("a"); }
static void b() { System.out.print("B"); }
// ใน main:
a(); b();
ดูคำตอบ

ABaB — a พิมพ์ A, เรียก b พิมพ์ B, กลับมาพิมพ์ a, แล้ว main เรียก b อีกครั้งพิมพ์ B

7.2 พารามิเตอร์และ argument

parameter คือตัวแปรในวงเล็บของเมธอดที่รอรับค่า ส่วน argument คือค่าจริงที่ส่งเข้าไปตอนเรียก Java คัดลอก argument ไปใส่ parameter ตามลำดับตำแหน่ง

Java
static void greet(String name, int times) { ... }   // name, times = parameter
greet("JHope", 3);                                  // "JHope", 3   = argument
        │      └──→ times = 3
        └──────────→ name  = "JHope"

ตัวอย่าง 7.2.1เมธอดเดียว ใช้กับข้อมูลต่างกัน

GreetParam.java
public class GreetParam {
    public static void main(String[] args) {
        greet("JHope");                  // Hello, JHope!
        greet("Suga");                   // Hello, Suga!
        String friend = "RM";
        greet(friend);                   // ส่งค่าในตัวแปรก็ได้
    }

    static void greet(String name) {     // name รับค่าที่ส่งมา
        System.out.println("Hello, " + name + "!");
    }
}
ผลลัพธ์
Hello, JHope!
Hello, Suga!
Hello, RM!

หลาย parameter

จำนวน ชนิด และลำดับของ argument ต้องตรงกับ parameter และ argument เป็นนิพจน์ได้ ซึ่งจะคำนวณก่อนส่ง

ตัวอย่าง 7.2.2วาดเส้นด้วยอักขระและความยาวที่กำหนด

DrawLine.java
public class DrawLine {
    public static void main(String[] args) {
        drawLine('*', 10);               // **********
        drawLine('-', 4);                // ----
        drawLine('=', 2 * 6);            // 12 ตัว  (คำนวณ 2 * 6 ก่อนส่ง)
    }

    static void drawLine(char symbol, int length) {
        for (int i = 0; i < length; i++) {
            System.out.print(symbol);
        }
        System.out.println();
    }
}

โจทย์ฝึก 7.2-P3drawTriangle(int style, int length)★★ ปานกลาง

7.5 ประกอบโปรแกรมจากเมธอด

ออกแบบจากบนลงล่าง (top-down): เขียนงานเป็นข้อ ๆ ภาษาคน แต่ละข้อที่ทำงานเดียวกลายเป็นเมธอดหนึ่งตัว แล้วกำหนดว่าเมธอดต้องรู้อะไร (parameter) และให้อะไรกลับ (return type) สุดท้าย main เรียกตามลำดับเหมือนสารบัญ

ตัวอย่าง 7.5.1รายงานเกรดจากเมธอดหลายตัว

GradeReport.java
import java.util.Scanner;

public class GradeReport {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Name: ");
        String name = input.nextLine();
        System.out.print("Work Mid Final: ");
        double work = input.nextDouble();
        double mid = input.nextDouble();
        double fin = input.nextDouble();

        double total = totalScore(work, mid, fin);   // รวม → 83.0
        char grade = gradeOf(total);                 // ตัดเกรด → A
        printReport(name, total, grade);             // แสดงผล
    }

    static double totalScore(double work, double mid, double fin) {
        return work + mid + fin;
    }

    static char gradeOf(double total) {
        if (total >= 80) return 'A';
        if (total >= 70) return 'B';
        if (total >= 60) return 'C';
        if (total >= 50) return 'D';
        return 'F';
    }

    static void printReport(String name, double total, char grade) {   // void: หน้าที่คือแสดงผล
        System.out.println("-".repeat(24));
        System.out.printf("%-8s %s%n", "Name", name);
        System.out.printf("%-8s %.1f%n", "Total", total);
        System.out.printf("%-8s %c%n", "Grade", grade);
        System.out.printf("%-8s %s%n", "Result", grade == 'F' ? "Fail" : "Pass");
        System.out.println("-".repeat(24));
    }
}

ขั้นที่ 1: hint ตัดสินคำทาย

เริ่มจากเมธอดที่ง่ายที่สุด รับคำทายกับเลขลับ แล้ว return ข้อความ ทดสอบใน main ด้วยค่าที่รู้คำตอบครบทั้ง 3 กรณี ยังไม่ต้องมี Scanner

ขั้นที่ 2: askGuess ถามจนได้ค่าในช่วง

เมธอดนี้บอกช่วงให้ผู้เล่นเห็น อ่านคำทาย ถ้าอยู่นอกช่วงให้ถามใหม่ (while) จนได้ค่าที่ใช้ได้แล้วจึง return ส่ง Scanner ตัวเดียวจาก main เข้ามาเป็น parameter

ตัวอย่าง 7.6.2ทดสอบ askGuess ด้วยค่านอกช่วง

GuessStep2.java
import java.util.Scanner;

public class GuessStep2 {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int g = askGuess(input, 1, 20);
        System.out.println("You guessed " + g);           // 7
    }

    static int askGuess(Scanner input, int min, int max) {
        System.out.print("Guess a number from " + min + " to " + max + ": ");
        int guess = input.nextInt();
        while (guess < min || guess > max) {              // 25 และ 0 นอกช่วง → ถามใหม่
            System.out.print("  Out of range. Enter " + min + "-" + max + ": ");
            guess = input.nextInt();
        }
        return guess;                                     // ถึงตรงนี้ได้ = อยู่ในช่วงแน่นอน
    }
}

ขั้นที่ 3: playGame วนจนทายถูก

ประกอบสองเมธอดแรกเข้าในลูป นับครั้ง และปรับ min/max หลังทายผิด ขั้นนี้ยังไม่สุ่ม ส่งเลขลับ 13 เข้าไปตรง ๆ ผลจึงต้องตรงกับตารางการเล่นด้านบนทุกบรรทัด

ตัวอย่าง 7.6.3เล่นด้วยเลขลับที่รู้คำตอบ

GuessStep3.java
import java.util.Scanner;

public class GuessStep3 {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int tries = playGame(input, 13, 1, 20);           // เลขลับคงที่ ไว้ทดสอบ
        System.out.println("Tries: " + tries);            // 4
    }

    static int playGame(Scanner input, int secret, int min, int max) {
        int tries = 0;
        int guess = 0;                                    // 0 ไม่ใช่คำตอบ → เข้าลูปแน่นอน
        while (guess != secret) {
            guess = askGuess(input, min, max);            // เรียกเมธอดจากขั้นที่ 2
            tries++;
            System.out.println("  " + hint(guess, secret));   // เรียกเมธอดจากขั้นที่ 1
            if (guess < secret) min = guess + 1;          // น้อยไป → ขยับขอบล่าง
            else if (guess > secret) max = guess - 1;     // มากไป → ขยับขอบบน
        }
        return tries;
    }

    static int askGuess(Scanner input, int min, int max) {
        System.out.print("Guess a number from " + min + " to " + max + ": ");
        int guess = input.nextInt();
        while (guess < min || guess > max) {
            System.out.print("  Out of range. Enter " + min + "-" + max + ": ");
            guess = input.nextInt();
        }
        return guess;
    }

    static String hint(int guess, int secret) {
        if (guess < secret) return "Too low";
        if (guess > secret) return "Too high";
        return "Correct!";
    }
}

ขั้นที่ 4: สุ่มเลขลับ แล้วประกอบเป็นเกมจริง

ใช้คลาส Random (import java.util.Random;) สุ่มจำนวนเต็ม: rng.nextInt(n) คืน 0 ถึง n − 1 ดังนั้นช่วง min–max เขียนเป็น min + rng.nextInt(max - min + 1) (รายละเอียดอยู่ในบทเสริม Random)

ระหว่างทดสอบใส่ seed เช่น new Random(7) จะได้เลขลับเดิมทุกครั้งที่รัน (seed 7 ได้ 17) เมื่อเล่นจริงให้ลบ seed ออกเป็น new Random()

ตัวอย่าง 7.6.4เกมทายตัวเลขฉบับสมบูรณ์

GuessNumberGame.java
import java.util.Random;
import java.util.Scanner;

public class GuessNumberGame {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Random rng = new Random(7);                       // ทดสอบ: seed 7 → 17 ทุกครั้ง (เล่นจริงใช้ new Random())
        final int MIN = 1, MAX = 20;

        System.out.println("=== Mystery Box: guess the number ===");
        int secret = randomSecret(rng, MIN, MAX);
        int tries = playGame(input, secret, MIN, MAX);
        System.out.println("The secret number was " + secret + ". You played " + tries + " times.");
    }

    static int randomSecret(Random rng, int min, int max) {
        return min + rng.nextInt(max - min + 1);          // nextInt(20) ได้ 0-19 → +1 ได้ 1-20
    }

    static int playGame(Scanner input, int secret, int min, int max) {
        int tries = 0;
        int guess = 0;
        while (guess != secret) {
            guess = askGuess(input, min, max);
            tries++;                                      // ค่านอกช่วงไม่ถูกนับ เพราะ askGuess ไม่คืนค่านั้น
            System.out.println("  " + hint(guess, secret));
            if (guess < secret) min = guess + 1;
            else if (guess > secret) max = guess - 1;
        }
        return tries;
    }

    static int askGuess(Scanner input, int min, int max) {
        System.out.print("Guess a number from " + min + " to " + max + ": ");
        int guess = input.nextInt();
        while (guess < min || guess > max) {
            System.out.print("  Out of range. Enter " + min + "-" + max + ": ");
            guess = input.nextInt();
        }
        return guess;
    }

    static String hint(int guess, int secret) {
        if (guess < secret) return "Too low";
        if (guess > secret) return "Too high";
        return "Correct!";
    }
}

7.7 ลงมือเขียน: เกมเป่ายิ้งฉุบกับคอมพิวเตอร์

ฝึกกระบวนการเดียวกับหัวข้อ 7.6 อีกครั้งกับเกมใหม่: อ่านโจทย์ → เขียนรหัสเทียม → เลือกเมธอด → เขียนและทดสอบทีละขั้น

โจทย์

  • คอมพิวเตอร์สุ่มเลือกหนึ่งในสามอย่าง
  • ค้อนชนะกรรไกร, กรรไกรชนะกระดาษ, กระดาษชนะค้อน ถ้าเลือกเหมือนกันคือเสมอ
  • ผู้ชนะในแต่ละรอบได้ 1 แต้ม เสมอไม่มีใครได้ ใครถึง 3 แต้มก่อนชนะทั้งแมตช์
  • ทุกรอบแสดงสิ่งที่ทั้งสองฝ่ายเลือก ผลของรอบ และคะแนน

    }

RockPaperScissors.java
import java.util.Random;
import java.util.Scanner;

public class RockPaperScissors {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Random rng = new Random(7);                       // ทดสอบ (เล่นจริงใช้ new Random())
        final int TARGET = 3;
        System.out.println("=== Rock Paper Scissors: first to " + TARGET + " wins ===");
        String winner = playMatch(input, rng, TARGET);
        System.out.println(winner + " won the match!");
    }

    static String playMatch(Scanner input, Random rng, int target) {
        int you = 0, com = 0, round = 0;
        while (you < target && com < target) {            // ยังไม่มีใครถึงเป้า
            round++;
            System.out.println("Round " + round);
            int p = askChoice(input);
            int c = computerChoice(rng);
            System.out.println("  You: " + nameOf(p) + "  Computer: " + nameOf(c));
            int result = judge(p, c);
            if (result == 1) {
                you++;
                System.out.println("  You win this round");
            } else if (result == -1) {
                com++;
                System.out.println("  Computer wins this round");
            } else {
                System.out.println("  Draw");                 // ไม่มีใครได้แต้ม
            }
            System.out.println("  Score " + you + " : " + com);
        }
        return you == target ? "You" : "Computer";
    }

    static int askChoice(Scanner input) {
        System.out.print("  1=Rock 2=Paper 3=Scissors: ");
        int choice = input.nextInt();
        while (choice < 1 || choice > 3) {
            System.out.print("  Please enter 1-3: ");
            choice = input.nextInt();
        }
        return choice;
    }

    static int computerChoice(Random rng) {
        return 1 + rng.nextInt(3);
    }

    static String nameOf(int choice) {
        if (choice == 1) return "Rock";
        if (choice == 2) return "Paper";
        return "Scissors";
    }

    static int judge(int player, int computer) {
        if (player == computer) return 0;
        if ((player == 1 && computer == 3)
                || (player == 2 && computer == 1)
                || (player == 3 && computer == 2)) return 1;
        return -1;
    }
}
