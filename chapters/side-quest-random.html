บทเสริม · Random

สุ่มตัวเลขและเกมทายคำตอบ

สร้างค่าที่เปลี่ยนได้ตามการรัน แล้วนำไปฝึกเงื่อนไข ลูป และการรับข้อมูลด้วยเกมทายตัวเลข

เมื่อจบบทนี้ นิสิตจะทำได้
  • เข้าใจช่วงค่าที่ Math.random คืน
  • สุ่มจำนวนเต็มในช่วงที่กำหนด
  • ใช้ Random object และ seed เพื่อทดสอบได้
  • ออกแบบ loop เกมทายเลขพร้อมคำใบ้

สุ่มค่าอย่างตั้งใจ

ก่อนเขียนสูตร ให้ระบุขอบเขตล่างและบนก่อนว่าแต่ละค่ารวมอยู่ในช่วงหรือไม่ ความผิดพลาดเรื่อง inclusive/exclusive เป็นบั๊กที่พบบ่อย

เกี่ยวกับ output ของการสุ่ม

ตัวอย่างที่ใช้ Math.random() หรือ new Random() แบบไม่กำหนด seed จะได้ผลต่างกันทุกครั้งที่รัน ผลที่แสดงเป็นเพียงการรันครั้งหนึ่ง ส่วนตัวอย่างที่ใช้ new Random(seed) จะได้ผลเหมือนกันทุกครั้ง จึงใช้ทดสอบได้

S1.1 Math.random และช่วง [0, 1)

Math.random() คืนค่า double ตั้งแต่ 0.0 รวม แต่ต่ำกว่า 1.0 (ไม่รวม) — เขียนแบบคณิตศาสตร์ว่า [0, 1)

สร้างสูตรสุ่มจำนวนเต็ม min ถึง max (รวมทั้งสองขอบ)
  1. นับจำนวนค่าที่เป็นไปได้: count = max - min + 1 (1–6 มี 6 ค่า)
  2. คูณ: Math.random() * count ได้ช่วง [0, count)
  3. cast เป็น int ตัดเศษ: ได้ 0, 1, …, count − 1
  4. บวก min: ได้ min, …, max
  5. สูตร: int v = min + (int) (Math.random() * (max - min + 1));

ตัวอย่าง S1.1.1ค่าดิบจาก Math.random

แนวคิด: ดูหน้าตาค่าที่ได้ — ทุกค่าอยู่ระหว่าง 0 ถึงน้อยกว่า 1

RawRandom.java
public class RawRandom {
    public static void main(String[] args) {
        for (int i = 1; i <= 5; i++) {
            // ค่าในช่วง [0, 1)
            double r = Math.random();
            // ×6 ได้ [0, 6) → (int) ได้ 0-5 → +1 ได้ 1-6
            System.out.printf("r = %.4f   r*6 = %.4f   (int)(r*6)+1 = %d%n", r, r * 6, (int) (r * 6) + 1);
        }
    }
}

อธิบายทีละขั้น

  2. r × 6 อยู่ใน [0, 6)
  3. (int) ตัดเศษ → 0–5 แล้ว + 1 → 1–6
  4. รันซ้ำจะได้ตัวเลขต่างจากนี้

ตัวอย่าง S1.1.2ทอยลูกเต๋า 6000 ครั้ง ตรวจว่ากระจายเท่ากัน

แนวคิด: ใช้อาเรย์นับความถี่ (บทที่ 8) แต่ละหน้าควรออกใกล้ 1000 ครั้ง ถ้าหน้า 6 ไม่เคยออกแปลว่าสูตรผิด

DiceFrequency.java
public class DiceFrequency {
    public static void main(String[] args) {
        int[] count = new int[7];  // ใช้ช่อง 1-6 (ช่อง 0 ไม่ใช้)
        for (int i = 0; i < 6000; i++) {
            // สุ่มลูกเต๋า 1-6
            int roll = (int) (Math.random() * 6) + 1;
            // นับจำนวนครั้งของแต่ละหน้า
            count[roll]++;
        }
        for (int face = 1; face <= 6; face++) {
            // # หนึ่งตัวแทน 50 ครั้ง
            System.out.printf("%d: %4d %s%n", face, count[face], "#".repeat(count[face] / 50));
        }
    }
}

ทดลองบั๊ก S1.1.3ลืม +1 หรือวาง cast ผิดที่

แนวคิด: สองบั๊กที่พบบ่อย: ไม่บวก 1 (ได้ 0–5) และ cast ก่อนคูณ (ได้ 0 เสมอ)

RandomBugs.java
public class RandomBugs {
    public static void main(String[] args) {
        int min1 = 99, max1 = -1, min2 = 99, max2 = -1;
        for (int i = 0; i < 10000; i++) {
            // วงเล็บ: คูณก่อนแล้วจึง cast
            int a = (int) (Math.random() * 6);
            // cast ทำก่อน: (int) ของ [0,1) ได้ 0 เสมอ
            int b = (int) Math.random() * 6;
            min1 = Math.min(min1, a); max1 = Math.max(max1, a);
            min2 = Math.min(min2, b); max2 = Math.max(max2, b);
        }
        System.out.println("(int)(Math.random() * 6) range: " + min1 + " to " + max1);
        System.out.println("(int) Math.random() * 6 range: " + min2 + " to " + max2);
    }
}

ตัวอย่าง S1.2.1สุ่มจำนวนเต็มในช่วงด้วย nextInt

แนวคิด: สร้าง Random หนึ่งตัวแล้วเรียกซ้ำ — nextInt(11) + 20 ได้ 20–30

RandomRange.java
import java.util.Random;

public class RandomRange {
    public static void main(String[] args) {
        Random rng = new Random();  // ตัวสุ่มของ java.util
        for (int i = 0; i < 8; i++) {
            // nextInt(11) ได้ 0-10 → +20 ได้ 20-30
            System.out.print((rng.nextInt(11) + 20) + " ");
        }
        System.out.println();
        // สุ่ม true/false
        System.out.println("coin: " + (rng.nextBoolean() ? "heads" : "tails"));
        // กำหนดช่วง [1, 7) ได้ตรง ๆ
        System.out.println("1-6 (Java 17+): " + rng.nextInt(1, 7));
    }
}

  2. nextBoolean จำลองการโยนเหรียญ
  3. nextInt(1, 7) → ขอบบนไม่รวม จึงได้ 1–6

ตัวอย่าง S1.2.2seed: สุ่มแบบทำซ้ำได้

แนวคิด: new Random(42) ให้ลำดับตัวเลขเดิมทุกครั้งที่รัน — ใช้ทดสอบโปรแกรมที่มีการสุ่ม และ output ด้านขวาจึงเหมือนกับที่นิสิตรันได้

SeededRandom.java
import java.util.Random;

public class SeededRandom {
    public static void main(String[] args) {
        // seed เดียวกัน → ลำดับเลขสุ่มเหมือนกัน
        Random a = new Random(42);
        Random b = new Random(42);
        Random c = new Random(7);  // seed ต่าง → ลำดับต่าง
        for (int i = 0; i < 6; i++) {
            System.out.printf("a=%d  b=%d  c=%d%n", a.nextInt(100), b.nextInt(100), c.nextInt(100));
        }
    }
}

a=84  b=84  c=44

อธิบายทีละขั้น
  1. a และ b ใช้ seed เดียวกัน → ได้ลำดับเดียวกันทุกค่า
  2. c ใช้ seed ต่าง → ลำดับต่าง
  3. ตัวเลขเหล่านี้จะเหมือนเดิมทุกครั้งที่รัน

ตัวอย่าง S1.2.3สุ่มเลือกจากอาเรย์และสับไพ่

แนวคิด: สุ่ม index แทนการสุ่มค่าโดยตรง: arr[rng.nextInt(arr.length)] และใช้ Collections.shuffle สับลำดับรายการ

PickAndShuffle.java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Random;

public class PickAndShuffle {
    public static void main(String[] args) {
        // ใส่ seed เพื่อให้ผลเหมือนเดิมทุกครั้ง
        Random rng = new Random(2026);
        String[] menu = {"Pad Thai", "Fried rice", "Som tam", "Khao man gai", "Noodle soup"};
        // สุ่ม index 0 ถึง length-1
        System.out.println("Today's lunch: " + menu[rng.nextInt(menu.length)]);

        List<String> students = new ArrayList<>(List.of("Mali", "Beam", "Nida", "Ploy", "Tan", "Pim"));
        Collections.shuffle(students, rng);  // สลับลำดับแบบสุ่ม
        System.out.println("Shuffled: " + students);
        // subList(0, 3) = index 0-2
        System.out.println("Team A: " + students.subList(0, 3));
        System.out.println("Team B: " + students.subList(3, 6));
    }
}
ผลลัพธ์
Today's lunch: Noodle soup
Shuffled: [Beam, Pim, Ploy, Mali, Nida, Tan]
Team A: [Beam, Pim, Ploy]
Team B: [Mali, Nida, Tan]
อธิบายทีละขั้น
  1. nextInt(menu.length) → index 0–4 ถูกต้องเสมอแม้เปลี่ยนจำนวนเมนู
  2. shuffle(list, rng) ใช้ seed ทำให้ผลทำซ้ำได้
  3. subList แบ่งทีมหลังสับแล้ว

ดูคำตอบ

nextInt(5) ได้ 0–4 → × 2 + 1 → 1, 3, 5, 7, 9 (เลขคี่ 1–9)

S1.3 เกมทายตัวเลข

ออกแบบเกม
  1. สุ่มคำตอบ 1–100 หนึ่งครั้งก่อนเริ่มลูป
  2. วน: รับคำทาย → นับครั้ง → เทียบกับคำตอบ → บอก Too high/Too low
  3. หยุดเมื่อทายถูก แล้วสรุปจำนวนครั้ง
  4. เพิ่มเติม: จำกัดจำนวนครั้ง, ตรวจค่านอกช่วง, เล่นใหม่

ตัวอย่าง S1.3.1เกมพื้นฐาน (ใช้ seed เพื่อให้ทดสอบได้)

แนวคิด: while วนจนทายถูก — ใช้ seed 7 ทำให้คำตอบเหมือนเดิมทุกครั้ง จึงเขียน input ทดสอบล่วงหน้าได้ (ในเกมจริงใช้ new Random())

GuessGame.java
import java.util.Random;
import java.util.Scanner;

public class GuessGame {
    public static void main(String[] args) {
        Random rng = new Random(7);
        int secret = rng.nextInt(100) + 1;  // สุ่มเลข 1-100

            tries++;

            else if (guess < secret) System.out.println("  Too low");
        }
        System.out.println("Correct! " + secret + " in " + tries + " tries");
    }
}

        final int MAX_TRIES = 5;
        // สร้าง Random แล้วสุ่มในบรรทัดเดียว

        Scanner input = new Scanner(System.in);
        boolean won = false;
        for (int t = 1; t <= MAX_TRIES; t++) {
            System.out.print("Try " + t + "/" + MAX_TRIES + ": ");
            int g = input.nextInt();

                System.out.println("  Out of range, not counted");
                t--;  // ลด t เพื่อชดเชย t++ ของลูป

                won = true;
                System.out.println("  Correct!");
                break;
            }
            System.out.println(g > secret ? "  Too high" : "  Too low");
        }
        // ใช้ครบทุกครั้งแล้วยังไม่ถูก
        if (!won) System.out.println("Out of tries. The number was " + secret);
    }
}

import java.util.Random;

public class RandomTesting {
    public static void main(String[] args) {
        // ส่วนที่ไม่สุ่ม ทดสอบด้วยค่าที่กำหนด
        System.out.println(hint(30, 42) + " / " + hint(50, 42) + " / " + hint(42, 42));

        Random rng = new Random();
        // ค่าเริ่มที่ทำให้ค่าแรกชนะเสมอ
        int min = Integer.MAX_VALUE, max = Integer.MIN_VALUE;
        // จดว่าเคยสุ่มได้ค่าใดบ้าง
        boolean[] seen = new boolean[11];
        // สุ่มหลายครั้งเพื่อตรวจช่วง
        for (int i = 0; i < 10000; i++) {
            int v = randomInRange(rng, 5, 10);
            min = Math.min(min, v);
            max = Math.max(max, v);
            seen[v] = true;
        }
        boolean allSeen = true;
        // ต้องเคยเห็นครบทุกค่า 5-10
        for (int v = 5; v <= 10; v++) allSeen &= seen[v];
        System.out.println("range check: min=" + min + " max=" + max + " allValuesSeen=" + allSeen);
    }

    static String hint(int guess, int secret) {
        if (guess > secret) return "Too high";
        if (guess < secret) return "Too low";
        return "Correct";
    }

    // สุ่มช่วงปิด [min, max]
    static int randomInRange(Random rng, int min, int max) {
        // max - min + 1 = จำนวนค่าที่เป็นไปได้
        return min + rng.nextInt(max - min + 1);
    }
}

import java.util.Random;

public class CoinFlips {
    public static void main(String[] args) {
        Random rng = new Random(5);  // seed ทำให้ผลคงที่
        int heads = 0, longest = 0, run = 0;
        char prev = ' ';  // ' ' = ยังไม่มีผลก่อนหน้า
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 20; i++) {
            // ?: สุ่มหัวหรือก้อย
            char f = rng.nextBoolean() ? 'H' : 'T';
            sb.append(f);
            if (f == 'H') heads++;
            // เหมือนเดิม → ต่อยาว, ต่าง → เริ่มนับใหม่
            run = (f == prev) ? run + 1 : 1;
            // จำความยาวที่มากที่สุด
            longest = Math.max(longest, run);
            prev = f;
        }
        System.out.println(sb);
        System.out.println("Heads: " + heads + ", Tails: " + (20 - heads));
        System.out.println("Longest streak: " + longest);
    }
}

RockPaperScissors.java
import java.util.Random;
import java.util.Scanner;

public class RockPaperScissors {
    // index 0, 1, 2 แทนท่า
    static final String[] CHOICES = {"rock", "paper", "scissors"};

    public static void main(String[] args) {
        Random rng = new Random(3);
        Scanner in = new Scanner(System.in);
        int win = 0, lose = 0, draw = 0;
        while (true) {
            System.out.print("Your move: ");
            String move = in.next().toLowerCase();
            // พิมพ์ quit เพื่อออก
            if (move.equals("quit")) break;
            // แปลงท่าเป็น index (-1 = ไม่ถูกต้อง)
            int user = indexOf(move);
            if (user < 0) {
                System.out.println("  invalid move");
                continue;
            }
            int cpu = rng.nextInt(3);  // คอมพิวเตอร์สุ่มท่า
            // 1 = ชนะ, -1 = แพ้, 0 = เสมอ
            int r = judge(user, cpu);
            System.out.println("  CPU: " + CHOICES[cpu] + " -> " + (r > 0 ? "you win" : r < 0 ? "you lose" : "draw"));
            if (r > 0) win++; else if (r < 0) lose++; else draw++;
        }
        System.out.println("Win " + win + ", Lose " + lose + ", Draw " + draw);
    }

    static int indexOf(String move) {
        for (int i = 0; i < CHOICES.length; i++) if (CHOICES[i].equals(move)) return i;
        return -1;
    }

    static int judge(int user, int cpu) {
        if (user == cpu) return 0;
        // ท่าที่อยู่ถัดไปหนึ่งตำแหน่ง (วนรอบ) ชนะ
        return (user - cpu + 3) % 3 == 1 ? 1 : -1;
    }
}

(user − cpu + 3) % 3 == 1 คือผู้ใช้ชนะ เพราะในลำดับ rock(0) → paper(1) → scissors(2) ตัวที่อยู่ถัดไปหนึ่งตำแหน่ง (วนรอบ) ชนะตัวก่อนหน้า

PasswordGenerator.java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Random;

public class PasswordGenerator {
    // ตัดตัวที่สับสนง่ายอย่าง O, I ออก
    static final String UPPER = "ABCDEFGHJKLMNPQRSTUVWXYZ", LOWER = "abcdefghijkmnopqrstuvwxyz",
        DIGITS = "23456789", SPECIAL = "!@#$%";

    public static void main(String[] args) {
        Random rng = new Random(2026);
        for (int i = 0; i < 3; i++) {
            String pw = generatePassword(rng, 10);
            System.out.println(pw + "  valid=" + isValid(pw));
        }
    }

    static String generatePassword(Random rng, int length) {
        String all = UPPER + LOWER + DIGITS + SPECIAL;
        // รายการอักขระ ยังสลับลำดับได้
        List<Character> chars = new ArrayList<>();
        // รับประกันว่ามีครบทุกกลุ่มอย่างน้อย 1 ตัว
        for (String group : new String[]{UPPER, LOWER, DIGITS, SPECIAL}) chars.add(pick(rng, group));
        // เติมจนครบความยาว
        while (chars.size() < length) chars.add(pick(rng, all));
        // สลับไม่ให้กลุ่มเรียงตามลำดับเดิม
        Collections.shuffle(chars, rng);
        StringBuilder sb = new StringBuilder();
        for (char c : chars) sb.append(c);
        return sb.toString();
    }

    static char pick(Random rng, String s) {
        // สุ่มอักขระหนึ่งตัวจากข้อความ
        return s.charAt(rng.nextInt(s.length()));
    }

    static boolean isValid(String pw) {
        // ตรวจว่ามีครบทุกกลุ่ม
        boolean u = false, l = false, d = false, s = false;
        for (char c : pw.toCharArray()) {
            if (Character.isUpperCase(c)) u = true;
            else if (Character.isLowerCase(c)) l = true;
            else if (Character.isDigit(c)) d = true;
            else if (SPECIAL.indexOf(c) >= 0) s = true;
        }
        return pw.length() >= 8 && u && l && d && s;
    }
}

ตัดตัวอักษรที่สับสนง่ายอย่าง O/0, I/l/1 ออกจากกลุ่มอักขระ เพื่อให้ผู้ใช้อ่านรหัสได้ถูก

Exercise S1.5จำลองมอนติคาร์โล: ประมาณค่า π★★★ ท้าทาย

สุ่มจุด (x, y) ในสี่เหลี่ยมจัตุรัส [0, 1) × [0, 1) จำนวน N จุด นับจุดที่อยู่ในวงกลมหนึ่งในสี่ (x² + y² ≤ 1) แล้วประมาณ π ≈ 4 × (จุดในวง / N) ทำสำหรับ N = 100, 10,000 และ 1,000,000 ด้วย new Random(1) แสดงค่าประมาณ และความคลาดเคลื่อนจาก Math.PI

สิ่งที่ต้องทำ

  2. แสดงตาราง N, ค่าประมาณ, error ทศนิยม 5 ตำแหน่ง
  3. อธิบายว่าทำไม N มากขึ้นค่าจึงแม่นขึ้น
ตัวอย่างผลลัพธ์

       100    3.12000    0.02159
    10,000    3.13960    0.00199
 1,000,000    3.14336    0.00176

MonteCarloPi.java
import java.util.Random;

public class MonteCarloPi {
    public static void main(String[] args) {
        Random rng = new Random(1);
        System.out.printf("%10s %10s %10s%n", "N", "estimate", "error");
        // _ คั่นหลักให้อ่านง่าย
        for (int n : new int[]{100, 10_000, 1_000_000}) {
            double pi = estimatePi(rng, n);
            System.out.printf("%,10d %10.5f %10.5f%n", n, pi, Math.abs(pi - Math.PI));
        }
    }

    static double estimatePi(Random rng, int n) {
        int inside = 0;
        for (int i = 0; i < n; i++) {
            // จุดสุ่มในสี่เหลี่ยม [0,1) × [0,1)
            double x = rng.nextDouble(), y = rng.nextDouble();
            // อยู่ในวงกลมรัศมี 1
            if (x * x + y * y <= 1) inside++;
        }
        return 4.0 * inside / n;  // สัดส่วนในวง ≈ π/4
    }
}

MultiRoundGuess.java
import java.util.ArrayList;
import java.util.Random;
import java.util.Scanner;

public class MultiRoundGuess {
    public static void main(String[] args) {
        Random rng = new Random(11);
        Scanner in = new Scanner(System.in);
        // เก็บจำนวนครั้งของรอบที่ชนะ
        ArrayList<Integer> wins = new ArrayList<>();
        int rounds = 0;
        String again;
        do {  // เล่นอย่างน้อย 1 รอบ
            rounds++;
            System.out.println("== Round " + rounds + " ==");
            // คืนจำนวนครั้งที่ใช้ หรือ -1 ถ้าแพ้
            int result = playRound(in, rng.nextInt(100) + 1);
            if (result > 0) wins.add(result);
            System.out.print("Play again? (y/n) ");
            again = in.next();
        // ตอบ y จึงเล่นต่อ
        } while (again.equalsIgnoreCase("y"));
        System.out.println("Rounds: " + rounds + ", Wins: " + wins.size());
        if (!wins.isEmpty()) {
            int sum = 0, best = Integer.MAX_VALUE;
            // ผลรวมและค่าน้อยสุดในลูปเดียว
            for (int w : wins) { sum += w; best = Math.min(best, w); }
            System.out.printf("Average tries: %.2f, Best: %d%n", (double) sum / wins.size(), best);
        }
    }

    static int playRound(Scanner in, int secret) {
        for (int t = 1; t <= 7; t++) {  // ทายได้รอบละ 7 ครั้ง
            System.out.print("Guess " + t + ": ");
            int g = in.nextInt();
            if (g == secret) {
                System.out.println("  Correct in " + t);
                return t;
            }
            System.out.println(g > secret ? "  Too high" : "  Too low");
        }
        System.out.println("  Lost! It was " + secret);
        return -1;
    }
}
