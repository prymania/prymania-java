บทที่ 5 · ให้โปรแกรมเลือกทำงาน

เงื่อนไขและการตัดสินใจ

สอนให้โปรแกรมตอบสนองต่อข้อมูลที่แตกต่างกัน ด้วย boolean การเปรียบเทียบ และโครงสร้าง if หรือ switch

เมื่อจบบทนี้ นิสิตจะทำได้
  • อ่านผลนิพจน์เปรียบเทียบเป็น true หรือ false
  • รวมเงื่อนไขด้วย &&, || และ !
  • เลือกใช้ if/else กับ switch
  • จัดลำดับเงื่อนไขไม่ให้กรณีสำคัญถูกบัง

5. เปลี่ยนเงื่อนไขให้เป็นเส้นทางของโปรแกรม

เงื่อนไขคือนิพจน์ที่ให้ผลเป็น true หรือ false โปรแกรมใช้ผลนี้เลือกคำสั่งที่จะทำต่อ — เหมือนข้าวหลามตัดใน Flowchart บทที่ 1 บทนี้เริ่มจากการสร้างค่า boolean ไปจนถึงการเลือกทางหลายทาง

5.1 boolean และตัวดำเนินการเปรียบเทียบ

boolean เก็บได้แค่ true หรือ false การเปรียบเทียบ เช่น score >= 50 ให้ผลเป็นค่า boolean ซึ่งเก็บในตัวแปรหรือใช้ใน if ได้

ตัวดำเนินการความหมายตัวอย่าง (x = 7)ผล
==เท่ากับx == 7true
!=ไม่เท่ากับx != 7false
>มากกว่าx > 10false
<น้อยกว่าx < 10true
>=มากกว่าหรือเท่ากับx >= 7true
<=น้อยกว่าหรือเท่ากับx <= 6false

ตัวอย่าง 5.1.1ผลการเปรียบเทียบคือ true หรือ false

Comparisons.java
public class Comparisons {
    public static void main(String[] args) {
        int score = 72;
        boolean passed = score >= 50;                         // true
        boolean perfect = score == 100;                       // false
        System.out.println("passed = " + passed);
        System.out.println("perfect = " + perfect);
        System.out.println("score < 80 ? " + (score < 80));   // true   ต่อข้อความต้องมีวงเล็บ
        System.out.println("score != 72 ? " + (score != 72)); // false
    }
}
ผลลัพธ์
passed = true
perfect = false
score < 80 ? true
score != 72 ? false
= กับ ==

= คือการใส่ค่า ส่วน == คือการเปรียบเทียบ เขียน if (score = 100) จะคอมไพล์ไม่ผ่าน (int cannot be converted to boolean)

ดูตัวอย่าง: error เมื่อใช้ = แทน ==

ทดลอง error 5.1.2ใช้ = แทน ==

AssignInIf.java
public class AssignInIf {

        int score = 72;
        if (score = 100) {
            System.out.println("Perfect!");
        }
    }
}

เปรียบเทียบ String ต้องใช้ equals

== กับ String ตรวจว่าเป็นออบเจ็กต์เดียวกันหรือไม่ ไม่ได้ดูตัวอักษร ให้ใช้ .equals() หรือ .equalsIgnoreCase() ถ้าไม่สนตัวพิมพ์เล็ก-ใหญ่

ตัวอย่าง 5.1.3ผู้ใช้พิมพ์ admin

StringCompare.java
import java.util.Scanner;

public class StringCompare {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Username: ");
        String user = input.next();
        System.out.println("user == \"admin\"      -> " + (user == "admin"));        // false
        System.out.println("user.equals(\"admin\") -> " + user.equals("admin"));     // true
        System.out.println("equalsIgnoreCase(\"ADMIN\") -> " + user.equalsIgnoreCase("ADMIN"));
    }
}
ผลลัพธ์ · ตัวที่ขีดเส้นใต้คือค่าที่ผู้ใช้พิมพ์
Username: admin
user == "admin"      -> false
user.equals("admin") -> true
equalsIgnoreCase("ADMIN") -> true
คำแนะนำ

จำไว้: String ใช้ equals เสมอ และ double ห้ามใช้ == ตรง ๆ เพราะเก็บค่าแบบประมาณ ให้ตรวจว่า “ต่างกันน้อยมาก” แทน: Math.abs(a - b) < 1e-9

ตัวอย่างเพิ่มเติม: เปรียบเทียบ double อย่างปลอดภัย

ตัวอย่าง 5.1.40.1 + 0.2 ไม่เท่ากับ 0.3 พอดี

DoubleCompare.java
public class DoubleCompare {
    public static void main(String[] args) {
        double a = 0.1 + 0.2;
        double b = 0.3;
        System.out.println("a = " + a);                                         // 0.30000000000000004
        System.out.println("a == b ? " + (a == b));                             // false
        System.out.println("close enough ? " + (Math.abs(a - b) < 1e-9));       // true
    }
}
ผลลัพธ์
a = 0.30000000000000004
a == b ? false
close enough ? true
ลองคิดก่อน · ผลของนิพจน์

ถ้า int a = 5, b = 8; ผลของ a * 2 > b, a + 3 == b, b % a != 3 คืออะไร

ดูคำตอบ

10 > 8 → true; 8 == 8 → true; 8 % 5 = 3, 3 != 3 → false

5.2 รวมเงื่อนไขด้วย AND, OR และ NOT

ใช้รวมหลายเงื่อนไขเข้าด้วยกัน

ตัวดำเนินการชื่อเป็นจริงเมื่อ
&&ANDทั้งสองข้างเป็นจริง
||ORอย่างน้อยหนึ่งข้างเป็นจริง
!NOTกลับค่า จริง ↔ เท็จ

ตัวอย่าง 5.2.1ตรวจช่วงอายุ 18–25

RangeCheck.java
public class RangeCheck {
    public static void main(String[] args) {
        int age = 19;
        boolean inRange = age >= 18 && age <= 25;        // true    ต้องจริงทั้งสองข้าง
        boolean outOfRange = age < 18 || age > 25;       // false   จริงข้างเดียวก็พอ
        System.out.println("18-25? " + inRange);
        System.out.println("outside? " + outOfRange);
        System.out.println("not in range? " + !inRange); // false
    }
}
ผลลัพธ์
18-25? true
outside? false
not in range? false
ข้อผิดพลาดที่พบบ่อย

คณิตศาสตร์เขียน 18 ≤ age ≤ 25 ได้ แต่ Java ต้องแยกเป็นสองเงื่อนไขแล้วเชื่อมด้วย && เสมอ เขียน 18 <= age <= 25 จะคอมไพล์ไม่ผ่าน

ลำดับของ && และ ||

&& ทำก่อน || เสมอ เหมือนคูณก่อนบวก ถ้าต้องการลำดับอื่นหรือให้อ่านง่าย ให้ใส่วงเล็บ

ตัวอย่าง 5.2.2สิทธิ์ส่วนลด: วงเล็บเปลี่ยนความหมาย

DiscountRule.java
public class DiscountRule {
    public static void main(String[] args) {
        boolean isStudent = false;
        boolean hasPass = true;
        int age = 16;

        boolean ruleA = (isStudent || hasPass) && age >= 18;   // (false || true) && false → false
        boolean ruleB = isStudent || hasPass && age >= 18;     // false || (true && false) → false
        boolean ruleC = isStudent || (hasPass && age >= 18);   // เหมือน B แต่อ่านง่ายกว่า

        System.out.println("A = " + ruleA);
        System.out.println("B = " + ruleB);
        System.out.println("C = " + ruleC);
    }
}
ผลลัพธ์
A = false
B = false
C = false
ลองทำดู

เปลี่ยน isStudent = true แล้วดูว่า A กับ B ต่างกันอย่างไร (A ยังเป็น false เพราะอายุไม่ถึง แต่ B เป็น true)

Short-circuit: หยุดตรวจเมื่อรู้คำตอบแล้ว

ถ้าด้านซ้ายของ && เป็น false คำตอบต้องเป็น false แน่นอน Java จึงไม่ตรวจด้านขวาเลย ใช้ป้องกันการหารด้วยศูนย์ได้

ตัวอย่าง 5.2.3ตรวจ count ก่อนหาร

ShortCircuit.java
public class ShortCircuit {
    public static void main(String[] args) {
        int total = 100;
        int count = 0;
        if (count != 0 && total / count > 10) {      // count != 0 เป็น false → ไม่หารเลย
            System.out.println("Average above 10");
        } else {
            System.out.println("No data or average too low");
        }
    }
}
ผลลัพธ์
No data or average too low
ตัวอย่างเพิ่มเติม: เงื่อนไขปีอธิกสุรทิน

ปีอธิกสุรทิน = หารด้วย 4 ลงตัว และ ไม่ลงตัวด้วย 100 หรือ หารด้วย 400 ลงตัว (ตัวอย่างนี้ใช้อาเรย์และลูปวนทดสอบหลายปี ซึ่งจะเรียนในบทที่ 6 และ 8 ให้โฟกัสที่บรรทัด boolean leap)

ตัวอย่าง 5.2.4ปีอธิกสุรทิน

LeapYear.java
public class LeapYear {
    public static void main(String[] args) {
        int[] years = {2024, 2025, 1900, 2000};
        for (int year : years) {
            boolean leap = (year % 4 == 0 && year % 100 != 0) || year % 400 == 0;
            System.out.println(year + " leap? " + leap);
        }
    }
}
ผลลัพธ์
2024 leap? true
2025 leap? false
1900 leap? false
2000 leap? true

5.3 if, else if และ else

if ทำคำสั่งในบล็อกเมื่อเงื่อนไขเป็นจริง มี 3 รูปแบบตามจำนวนทางเลือก

รูปแบบใช้เมื่อ
if (...) { }ทำเพิ่มเฉพาะบางกรณี กรณีอื่นไม่ต้องทำอะไร
if (...) { } else { }มีสองทาง ทำทางใดทางหนึ่งเสมอ
if (...) { } else if (...) { } else { }หลายทาง ตรวจจากบนลงล่าง ทำเฉพาะทางแรกที่จริง

ตัวอย่าง 5.3.1if อย่างเดียว: แจ้งเตือนเมื่อมีไข้

FeverAlert.java
import java.util.Scanner;

public class FeverAlert {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Temperature: ");
        double temp = input.nextDouble();
        if (temp >= 37.5) {                              // 38.2 ≥ 37.5 → true

5.4 เลือกทางด้วย switch

switch เหมาะเมื่อเปรียบเทียบค่าเดียวกับตัวเลือกที่เป็นค่าแน่นอนหลายค่า เช่น รหัสเมนู วันในสัปดาห์ ใช้กับ int, char, String ได้ แต่ใช้กับช่วงค่า (≥ 80) ไม่ได้

สถานการณ์ควรใช้
ตรวจช่วงค่า (≥, <)if-else if
เงื่อนไขหลายตัวแปรรวมกัน (&&, ||)if
ค่าเดียวเทียบกับค่าคงที่หลายค่าswitch
เมนูตัวเลข / ตัวอักษรswitch

switch แบบลูกศร (Java 14+)

เขียนแบบ case ค่า -> คำสั่ง ไม่ต้องมี break ใส่หลายค่าใน case เดียวได้ และคืนค่าได้ (มี ; ปิดท้าย)

ตัวอย่าง 5.4.1ฤดูจากเดือน

Season.java
import java.util.Scanner;

public class Season {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Month (1-12): ");
        int month = input.nextInt();
        String season = switch (month) {          // switch คืนค่ามาเก็บในตัวแปร
            case 3, 4, 5 -> "Summer";             // 4 ตรง case นี้
            case 6, 7, 8, 9, 10 -> "Rainy";
            case 11, 12, 1, 2 -> "Winter";
            default -> "Invalid month";           // ไม่ตรง case ใดเลย
        };
        System.out.println("Season: " + season);
    }
}

import java.util.Scanner;

public class MiniCalculator {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.print("Expression (a op b): ");
        double a = input.nextDouble();
        char op = input.next().charAt(0);              // อักขระแรกของคำ
        double b = input.nextDouble();

        switch (op) {
            case '+' -> System.out.println("= " + (a + b));
            case '-' -> System.out.println("= " + (a - b));
            case '*', 'x' -> System.out.println("= " + (a * b));
            case '/' -> {                              // หลายคำสั่ง → ใช้ { }
                if (b == 0) System.out.println("Cannot divide by zero");
                else System.out.println("= " + (a / b));
            }
            default -> System.out.println("Unknown operator: " + op);
        }
    }
}

Exercise 5.10เครื่องคิดเลข BMI พร้อมคำแนะนำ★★★ ท้าทาย

รับเพศ (M/F) น้ำหนัก (กก.) และส่วนสูง (ซม.) คำนวณ BMI แล้วจัดกลุ่มตามเกณฑ์เอเชีย: ต่ำกว่า 18.5 Underweight, 18.5–22.9 Normal, 23–24.9 Overweight, 25–29.9 Obese I, ตั้งแต่ 30 Obese II

นอกจากนี้ให้คำนวณน้ำหนักที่ควรลด/เพิ่มเพื่อให้ BMI อยู่ในช่วงปกติ (ใช้ขอบ 18.5 หรือ 22.9) และถ้าข้อมูลไม่สมเหตุสมผล (น้ำหนัก ≤ 0, ส่วนสูง < 50 หรือ > 250, เพศไม่ใช่ M/F) ให้แสดง Invalid input แล้วไม่คำนวณต่อ

สิ่งที่ต้องทำ
  1. ตรวจความถูกต้องของข้อมูลก่อน
  2. เพศใช้ char ผ่าน next().toUpperCase().charAt(0)
  3. น้ำหนักเป้าหมาย = BMI ขอบ × (ส่วนสูงเมตร)²
  4. แสดงผลทศนิยม 1–2 ตำแหน่งตามตัวอย่าง

Exercise 5.11วันถัดไปในปฏิทิน★★★ ท้าทาย
