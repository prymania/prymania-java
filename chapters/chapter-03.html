3. เก็บข้อมูลและคำนวณด้วย Java

ตัวแปรเปรียบเหมือนกล่องที่มีชื่อและชนิด ค่าในกล่องเปลี่ยนได้ระหว่างที่โปรแกรมทำงาน แต่ชนิดข้อมูลจะกำหนดว่ากล่องนั้นเก็บอะไรได้และนำไปคำนวณแบบใด ในบทนี้ทุกตัวอย่างใช้ค่าที่กำหนดในโค้ด ส่วนการรับค่าจากผู้ใช้จะเรียนในบทที่ 4

3.1 ตัวแปรและการกำหนดค่า

ตัวแปรคือกล่องที่มีชื่อและชนิด ประกาศโดยบอกชนิดและชื่อ แล้วใช้ = ใส่ค่าลงไป

รูปแบบตัวอย่างความหมาย
ชนิด ชื่อ;int age;ประกาศอย่างเดียว ยังไม่มีค่า
ชื่อ = ค่า;age = 19;ใส่ค่าภายหลัง
ชนิด ชื่อ = ค่า;int age = 19;ประกาศพร้อมใส่ค่า (ใช้บ่อยที่สุด)

ตัวอย่าง 3.1.1ประกาศ ใส่ค่า และพิมพ์

FirstVariables.java
public class FirstVariables {
    public static void main(String[] args) {
        int age = 19;
        String name = "JHope";                    // ข้อความอยู่ใน "..."

        System.out.println(name);                 // JHope   ไม่มี "" → พิมพ์ค่าในตัวแปร
        System.out.println(age);                  // 19
        System.out.println("name");               // name    มี "" → พิมพ์คำนี้ตรงตัว
        System.out.println("Name: " + name + ", age " + age);   // + ต่อข้อความกับค่า
    }
}

= คือการใส่ค่า ไม่ใช่ “เท่ากับ”

= แปลว่า “คำนวณด้านขวาให้เสร็จ แล้วเก็บลงตัวแปรด้านซ้าย” ดังนั้น x = x + 1 จึงถูกต้อง ตัวแปรเก็บได้ทีละค่า ค่าใหม่จะเขียนทับค่าเก่า

ตัวอย่าง 3.1.2ค่าเปลี่ยนตามลำดับคำสั่ง

ChangingValues.java
public class ChangingValues {
    public static void main(String[] args) {
        int bookCount = 3;                        // 3
        System.out.println("Start: " + bookCount);
        bookCount = bookCount + 2;                // 3 + 2 → 5
        System.out.println("After +2: " + bookCount);
        bookCount = 10;                           // เขียนทับ → 10 (ค่า 5 หายไป)
        System.out.println("Reset: " + bookCount);
        bookCount = bookCount * 2 - 1;            // 10 × 2 - 1 → 19
        System.out.println("Final: " + bookCount);
    }
}

Final: 19
ตัวอย่างเพิ่มเติม: สลับค่าตัวแปรสองตัว

ถ้าเขียน a = b; b = a; ค่าเดิมของ a จะหายก่อน จึงต้องฝากค่าไว้ในตัวแปรชั่วคราว (temp) ก่อน

ตัวอย่าง 3.1.3สลับค่าด้วย temp

SwapValues.java
public class SwapValues {
    public static void main(String[] args) {
        int a = 5;
        int b = 8;
        System.out.println("Before: a=" + a + ", b=" + b);

        int temp = a;     // ฝาก 5 ไว้
        a = b;            // a = 8
        b = temp;         // b = 5

        System.out.println("After:  a=" + a + ", b=" + b);
    }
}

ชนิดข้อมูลบอกว่ากล่องเก็บอะไรได้และคำนวณแบบใด เลือกจากลักษณะของค่าที่จะเก็บ

int กับ double

ตัวอย่าง 3.2.2double แสดงจุดทศนิยมเสมอ

IntVsDouble.java
public class IntVsDouble {
    public static void main(String[] args) {
        int whole = 5;
        double decimal = 5;              // int ใส่ double ได้ → 5.0
        double price = 19.99;
        System.out.println(whole);       // 5
        System.out.println(decimal);     // 5.0
        System.out.println(price);       // 19.99
        System.out.println(0.1 + 0.2);   // 0.30000000000000004  double เก็บทศนิยมแบบประมาณ
    }
}

ตัวดำเนินการ (operator) คือเครื่องหมายที่สั่งให้คำนวณ เช่น บวก ลบ คูณ หาร ส่วนที่มือใหม่พลาดบ่อยคือ การหารจำนวนเต็ม และ ลำดับการคำนวณ

ตัวอย่าง 3.3.1ลองทุกตัวดำเนินการ

Arithmetic.java
public class Arithmetic {
    public static void main(String[] args) {
        int a = 17, b = 5;
        System.out.println(a + b);      // 22
        System.out.println(a - b);      // 12
        System.out.println(a * b);      // 85
        System.out.println(a / b);      // 3     int / int ได้ int (ตัดเศษทิ้ง)
        System.out.println(a % b);      // 2     เศษ: 17 - 5×3 = 2
        System.out.println(17.0 / 5);   // 3.4   มี double อย่างน้อยหนึ่งตัว
    }
}

public class Shorthand {
    public static void main(String[] args) {
        int points = 10;
        points += 5;   // 15   points = points + 5
        points -= 3;   // 12
        points *= 2;   // 24
        points /= 5;   // 4    ยังเป็นการหาร int

        count++;       // 1
        count++;       // 2
        count--;       // 1
        System.out.println("count = " + count);
    }
}

public class AverageTrap {
    public static void main(String[] args) {
        int total = 7;
        int count = 2;

        double wrong = total / count;                  // 3.0  int / int = 3 แล้วจึงเป็น 3.0
        double right1 = total / 2.0;                   // 3.5  หารด้วย 2.0
        double right2 = (double) total / count;        // 3.5  แปลง total ก่อนหาร
        double stillWrong = (double) (total / count);  // 3.0  หารเสร็จแล้วค่อยแปลง

        System.out.println("wrong      = " + wrong);
        System.out.println("right1     = " + right1);
        System.out.println("right2     = " + right2);
        System.out.println("stillWrong = " + stillWrong);
    }
}
